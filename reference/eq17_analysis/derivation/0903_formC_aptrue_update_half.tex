\section*{a$'_{\mathrm{true}}$ arm, 4-state: the update half-step}

(Companion of 0902; $\mathbf{x} = [\delta\bw_1, \delta\bw_2, \delta\bw_3, \ba_w]^{\top}$, $e = x - \hat{x}$;
stages inside call $k$: $\hat{x}^{-}$ after predict, $\hat{x}^{(1)}$ after the $y_1$ update, $\hat{x}$ after the $y_2$ update;
$\ell_{ij}$ = entry $(i,j)$ of $L[k]$; hold: $\Delta\bw_d = 0$, $\nabla_d\bw_d = 0$, $F_{dw} = 0$)

\section{Update half-step}

\begin{align*}
S_1[k] &= P^{-}_{11}[k] + R_1 , \qquad \ell_{i1}[k] = \frac{P^{-}_{i1}[k]}{S_1[k]} \\
e_{y_1}[k] &= y_1[k] - \hat{x}_1^{-}[k] = e_1^{-}[k] + n_w[k] \\
\hat{x}^{(1)}[k] &= \hat{x}^{-}[k] + \ell_{\cdot 1}[k]\,e_{y_1}[k] \\
e^{(1)}[k] &= e^{-}[k] - \ell_{\cdot 1}[k]\bigl(e_1^{-}[k] + n_w[k]\bigr) \\
P^{(1)}[k] &= \bigl(I - \ell_{\cdot 1}e_1^{\top}\bigr)P^{-}\bigl(I - \ell_{\cdot 1}e_1^{\top}\bigr)^{\top} + \ell_{\cdot 1}R_1\ell_{\cdot 1}^{\top}
\end{align*}

\begin{align*}
H_2[k] &= a_{\mathrm{cov}}(1-S)\,[\,0,\ 0,\ 0,\ 1\,] \qquad \text{(hold)} \\
S_2[k] &= H_2P^{(1)}H_2^{\top} + R_2 , \qquad \ell_{\cdot 2}[k] = \frac{P^{(1)}H_2^{\top}}{S_2} \\
e_{y_2}[k] &= y_2[k] - H_2\hat{x}^{(1)}[k] \\
\hat{x}[k] &= \hat{x}^{(1)}[k] + \ell_{\cdot 2}[k]\,e_{y_2}[k] \\
e[k] &= e^{(1)}[k] - \ell_{\cdot 2}[k]\,e_{y_2}[k]
\end{align*}

\begin{equation*}
\hat{\ba}_w[k] - \hat{\ba}_w^{-}[k] = \ell_{41}[k]\,e_{y_1}[k] + \ell_{42}[k]\,e_{y_2}[k]
\end{equation*}

\section{Lemma: $\ell_{41} = -\ba'\,\ell_{31}$ in the hold}

error dynamics of 0902 (hold, first order), and the law-consistent combination
\begin{align*}
e_3^{-}[k+1] &= \lambda_c\,e_3[k] - F_{dw}[k]\,e_4[k] - \varepsilon_w[k] \\
e_4^{-}[k+1] &= e_4[k] + \ba'[k]\bigl[(1-\lambda_c)\,e_3[k] + F_{dw}[k]\,e_4[k] + \varepsilon_w[k]\bigr] \\
e_4^{-}[k+1] + \ba'[k]\,e_3^{-}[k+1] &= e_4[k] + \ba'[k]\,e_3[k]
\end{align*}
\begin{align*}
\mathrm{Cov}\bigl(e_4 + \ba' e_3,\ e_3\bigr)^{-}[k+1] &= \lambda_c\,\mathrm{Cov}\bigl(e_4 + \ba' e_3,\ e_3\bigr)[k]
   - F_{dw}[k]\,\mathrm{Cov}\bigl(e_4 + \ba' e_3,\ e_4\bigr)[k] \\
\mathrm{Cov}\bigl(e_4 + \ba' e_3,\ e_1\bigr)^{-}[k+1] &= \mathrm{Cov}\bigl(e_4 + \ba' e_3,\ e_2\bigr)[k] \\
\mathrm{Cov}\bigl(e_4 + \ba' e_3,\ e_2\bigr)^{-}[k+1] &= \mathrm{Cov}\bigl(e_4 + \ba' e_3,\ e_3\bigr)[k]
\end{align*}
\begin{equation*}
F_{dw} = 0 \;\Rightarrow\; \mathrm{Cov}\bigl(e_4 + \ba' e_3,\ e_1\bigr)^{-} \to 0 \;\Rightarrow\;
P^{-}_{41} = -\ba'\,P^{-}_{31} \;\Rightarrow\; \ell_{41} = -\ba'\,\ell_{31}
\end{equation*}
(the combination $e_4 + \ba' e_3$ is untouched by the predict and, once its covariance with $e_1$ has
decayed as $\lambda_c^{k}$, by the $y_1$ update; $F_{dw} \neq 0$ sources it with
$-F_{dw}\mathrm{Var}(e_4 + \ba' e_3)$, which is where $\ell_{41} + \ba'\ell_{31} \neq 0$ in motion comes from;
the $y_2$ update sources $\mathrm{Cov}(e_4 + \ba' e_3,\ e_j)$ by $-(H_{24}^{2}P^{(1)}_{j4}/S_2)\,\mathrm{Cov}(e_4 + \ba' e_3,\ e_4)$,
a factor $\sim 10^{-5}$, so the relation holds to that order; the decay of $\mathrm{Cov}(e_4 + \ba' e_3,\ e_1)$ is set by the
predict-plus-$y_1$ closed loop, eigenvalue $\approx 0.41$, faster than $\lambda_c$;
log 2026-09-03: $|\ell_{41} + \ba'\ell_{31}| < 10^{-4}$ in every hold, both trajectories)

\section{State dependence of the gains}

\begin{align*}
\ba'[k] &= \ba'_{\mathrm{true}}\bigl(\bw_d[k-1] - \delta\hat{\bw}_3[k-1]\bigr)
        = \ba'_0 - \ba''\bigl(\delta\hat{\bw}_3[k-1] - \E[\delta\hat{\bw}_3]\bigr) + \mathcal{O}(e^{2}) \\
\ell_{41}[k] &= -\ba'[k]\,\ell_{31}[k]
        = \ell_{41,0} + \ba''\,\ell_{31}\bigl(\delta\hat{\bw}_3[k-1] - \E[\delta\hat{\bw}_3]\bigr) \\
\ell_{42}[k] &= \frac{P^{(1)}_{44}\,H_{24}}{H_{24}^{2}P^{(1)}_{44} + R_2(\hat{\ba}_w)} , \qquad
   H_{24} = a_{\mathrm{cov}}\bigl(1 - S(\hat{\ba}_w)\bigr) , \qquad
   R_2 = 2a_{\mathrm{cov}}^{2}\,IF_{\mathrm{var}}(\hat{\ba}_w)\,(\hat{\ba}_w + \xi)^{2} \\
\ell_{42}[k] &= \ell_{42,0} + \frac{\partial\ell_{42}}{\partial\hat{\ba}_w}\bigl(\hat{\ba}^{(1)}_w[k] - \E[\hat{\ba}_w]\bigr) \\
\frac{\partial\ell_{42}}{\partial\hat{\ba}_w}
  &= \frac{\ell_{42}}{H_{24}}\Bigl(1 - \frac{2H_{24}^{2}P^{(1)}_{44}}{S_2}\Bigr)\frac{\partial H_{24}}{\partial\hat{\ba}_w}
  - \frac{\ell_{42}}{S_2}\,\frac{\partial R_2}{\partial\hat{\ba}_w}
\end{align*}

\section{Process/measurement noise cross-covariance $M$}

\begin{align*}
w[k] &:= x[k+1] - F[k]\,x[k] - B[k]\,u[k] = \varepsilon_w[k]\,[\,0,\ 0,\ -1,\ \ba'[k]\,]^{\top} \\
\varepsilon_w[k] &= \ba_w[k]\bar{f}_T[k] + (1-\lambda_c)\sum_{i=1}^{d}\ba_w[k-i]\bar{f}_T[k-i] + (1-\lambda_c)\,n_w[k] \\
y_1[k] &= \delta\bw_1[k] + n_w[k] \\
Q[k] &= \E\bigl[w w^{\top}\bigr] = Q_{33}[k]\,[\,0,\ 0,\ -1,\ \ba'\,]^{\top}[\,0,\ 0,\ -1,\ \ba'\,] \\
Q_{33} &= \kappa_T\Bigl(\hat{\ba}_w + (1-\lambda_c)^{2}\sum_{i=1}^{d}\hat{\ba}_w[k-i]\Bigr) + (1-\lambda_c)^{2}R_1 \\
M[k] &:= \E\bigl[w[k]\,n_w[k]\bigr] = R_1\,g_n[k] , \qquad g_n[k] := (1-\lambda_c)\,[\,0,\ 0,\ -1,\ \ba'[k]\,]^{\top} \\
\frac{M M^{\top}}{R_1} &= (1-\lambda_c)^{2}R_1\,[\,0,\ 0,\ -1,\ \ba'\,]^{\top}[\,0,\ 0,\ -1,\ \ba'\,] \qquad \text{(the $n_w$ share of $Q$)}
\end{align*}

\section{Mean of the update half-step}

\begin{align*}
\E\bigl[\hat{\ba}_w[k] - \hat{\ba}_w^{-}[k]\bigr]
  &= \ell_{41,0}\,\E\bigl[e_{y_1}[k]\bigr] + \ell_{42,0}\,\E\bigl[e_{y_2}[k]\bigr] \\
  &\quad + \ba''\,\ell_{31}\,\E\Bigl[\bigl(\delta\hat{\bw}_3[k-1] - \E[\delta\hat{\bw}_3]\bigr)\,e_{y_1}[k]\Bigr] \\
  &\quad + \frac{\partial\ell_{42}}{\partial\hat{\ba}_w}\,\E\Bigl[\bigl(\hat{\ba}^{(1)}_w[k] - \E[\hat{\ba}_w]\bigr)\,e_{y_2}[k]\Bigr]
\end{align*}

the two covariances through the estimate/error cross-moment $C := \E[\hat{x}\,e^{\top}]$ (posterior)
\begin{align*}
\E\Bigl[\delta\hat{\bw}_3[k-1]\,e_{y_1}[k]\Bigr] &= \E\Bigl[\hat{x}_3[k-1]\bigl(e_1^{-}[k] + n_w[k]\bigr)\Bigr]
   = \E\Bigl[\hat{x}_3[k-1]\,e_2[k-1]\Bigr] = C_{32}[k-1] \\
\E\Bigl[\hat{\ba}^{(1)}_w[k]\,e_{y_2}[k]\Bigr] &= \bigl(C^{(1)}[k]\,H_2^{\top}\bigr)_4
\end{align*}
recursion of $C$ under the filter's gains, true noise statistics ($M \neq 0$)
\begin{align*}
\E\bigl[\hat{x}[k]\,n_w[k]\bigr] &= \bigl(I - \ell_{\cdot 2}H_2\bigr)\ell_{\cdot 1}[k]\,R_1 , \qquad
\E\bigl[\hat{x}[k]\,w[k]^{\top}\bigr] = \bigl(I - \ell_{\cdot 2}H_2\bigr)\ell_{\cdot 1}[k]\,M^{\top} \\
C^{-}[k+1] &= F\,C[k]\,F^{\top} + F\bigl(I - \ell_{\cdot 2}H_2\bigr)\ell_{\cdot 1}[k]\,M^{\top} \\
C^{(1)}[k] &= C^{-}[k]\bigl(I - \ell_{\cdot 1}e_1^{\top}\bigr)^{\top}
   + \ell_{\cdot 1}\,e_1^{\top}P^{-}_{\mathrm{true}}[k]\bigl(I - \ell_{\cdot 1}e_1^{\top}\bigr)^{\top}
   - \ell_{\cdot 1}R_1\ell_{\cdot 1}^{\top} \\
C[k] &= C^{(1)}[k]\bigl(I - \ell_{\cdot 2}H_2\bigr)^{\top}
   + \ell_{\cdot 2}\,H_2P^{(1)}_{\mathrm{true}}[k]\bigl(I - \ell_{\cdot 2}H_2\bigr)^{\top}
   - \ell_{\cdot 2}R_2\ell_{\cdot 2}^{\top} \\
P^{-}_{\mathrm{true}}[k+1] &= F\,P_{\mathrm{true}}[k]\,F^{\top} + Q
   - F\bigl(I - \ell_{\cdot 2}H_2\bigr)\ell_{\cdot 1}[k]\,M^{\top}
   - M\,\ell_{\cdot 1}[k]^{\top}\bigl(I - \ell_{\cdot 2}H_2\bigr)^{\top}F^{\top}
\end{align*}
\begin{equation*}
M = 0 \;\Rightarrow\; P_{\mathrm{true}} = P ,\quad C \equiv 0 ,\quad
\text{both product terms} = 0 \qquad (\text{innovation} \perp \text{past})
\end{equation*}
