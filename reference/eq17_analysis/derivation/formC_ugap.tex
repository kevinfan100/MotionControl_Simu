\section{Coordinate and law}

(bars denote normalized quantities: lengths by $R$, $\ba = a/a_o$,
$\bar{f} = a_o f$; $0 < \ba < 1 \Leftrightarrow u > 1$;
$e_{\theta} = \theta - \hat{\theta}$)

\begin{equation*}
u := \frac{1}{1-\ba},
\qquad
\ba = 1 - \frac{1}{u},
\qquad
a_w = a_o\,\ba_w ,
\qquad
a_o = \frac{\Delta t}{\gamma_N R}
\end{equation*}

\begin{equation*}
\frac{\mathrm{d}\ba}{\mathrm{d}\bw} = \bigl(1-\ba\bigr)^{2}\bigl(1+\delta a\bigr)
\qquad\Longleftrightarrow\qquad
\frac{\mathrm{d}u}{\mathrm{d}\bw} = 1 + \delta a
\end{equation*}

\begin{equation*}
u(\bw) = \bigl(1+\delta a\bigr)\bigl(\bw - \bw_0\bigr),
\qquad
\frac{\partial u}{\partial \bw_0} = -\bigl(1+\delta a\bigr),
\qquad
\int_{\bw[k]}^{\bw[k+1]}\!\!\mathrm{d}u
 = \bigl(1+\delta a\bigr)\,\Sigma[k] \quad(\text{exact; } \Sigma \text{ in S3})
\end{equation*}

\section{Jacobians}

\begin{equation*}
\frac{\partial \ba}{\partial u} = \frac{1}{u^{2}} = \bigl(1-\ba\bigr)^{2},
\qquad
\frac{\partial}{\partial u}\Bigl[u + (1+\delta a)\bigl(\,\cdot\,\bigr)\Bigr] = 1,
\qquad
\frac{\partial}{\partial\,\delta a}
  \Bigl[u + (1+\delta a)\,\Sigma\Bigr] = \Sigma
\end{equation*}

\section{True dynamics}

$\bigl(\delta\bw_j[k] := \delta\bw[k-d+j-1],\ j = 1,\dots,d+1,\ d = 2\bigr)$

\begin{align*}
\bw[k] &= \bw_d[k] - \delta\bw_3[k], &
\bw_d[k+1] &= \bw_d[k] + \Delta\bw_d[k] \\
\delta\bw_1[k+1] &= \delta\bw_2[k], &
\delta\bw_2[k+1] &= \delta\bw_3[k] \\
\delta\bw_3[k+1] &= \lambda_c\,\delta\bw_3[k] - F_{dw}[k]\,e_{\ba_w}[k]
  - \varepsilon_w[k]
\end{align*}

\begin{align*}
F_{dw}[k] &= \bar{f}_{dw}[k] + (1-\lambda_c)\sum_{i=1}^{d}\bar{f}_{dw}[k-i] \\
\varepsilon_w[k] &= \ba_w[k]\,\bar{f}_T[k]
  + (1-\lambda_c)\sum_{i=1}^{d}\ba_w[k-i]\,\bar{f}_T[k-i]
  + (1-\lambda_c)\,n_w[k]
\end{align*}

\begin{equation*}
\Sigma[k] := \Delta\bw_d[k] + (1-\lambda_c)\,\delta\bw_3[k]
  + F_{dw}[k]\,e_{\ba_w}[k] + \varepsilon_w[k]
  \;=\; \bw[k+1] - \bw[k]
\end{equation*}

\begin{equation*}
\text{(a)}\quad u[k+1] = u[k] + \Sigma[k],
\qquad
\text{(b)}\quad u[k+1] = u[k] + \bigl(1+\delta a[k]\bigr)\Sigma[k],
\qquad
\delta a[k+1] = \delta a[k]
\end{equation*}
