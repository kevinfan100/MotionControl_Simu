\section{Gain law}

(bars denote normalized quantities: lengths by $R$; $a_o := \Delta t/(\gamma_N R)$ [1/pN] and $a_{\mathrm{nom}} := \Delta t/\gamma_N = a_o R$ [$\mu$m/pN];
$\ba := a/a_{\mathrm{nom}}$; forces by $a_o$: $\bar f := a_o f$, so that $\ba\,\bar f = \dW$; $0 \le \ba \le 1$, far field 1, contact 0;
hats sit on the normalized quantities $\ah$, $\bh$, $\delta\hat{\bw}$; $e_\theta = \theta - \hat\theta$)

\begin{align*}
\ba(\bw)      &= 1 - \frac{1}{b\,(\bw - \bw_s)} \\
\ap(\ba, b)   &:= \frac{d\ba}{d\bw} = b\bigl(1-\ba\bigr)^{2} \\
\app(\ba, b)  &:= \frac{d\ap}{d\bw} = -2\,b^{2}\bigl(1-\ba\bigr)^{3} \\
a_w &= a_o\,\ba_w , \qquad a_o = \frac{\Delta t}{\gamma_N R}
\end{align*}

\section{True dynamics}

$\bigl(\delta\bw_j[k] := \delta\bw[k-d+j-1],\ j = 1,\dots,d+1,\ d = 2\bigr)$

\begin{align*}
\bw[k]           &= \bw_d[k] - \delta\bw_3[k] \\
\bw_d[k+1]       &= \bw_d[k] + \Delta\bw_d[k] \\
\delta\bw_1[k+1] &= \delta\bw_2[k] \\
\delta\bw_2[k+1] &= \delta\bw_3[k] \\
\delta\bw_3[k+1] &= \lambda_c\,\delta\bw_3[k] - F_{dw}[k]\,e_{\ba_w}[k] - \varepsilon_w[k] \\
\dW[k]           &= \Delta\bw_d[k] + (1-\lambda_c)\,\delta\bw_3[k] + F_{dw}[k]\,e_{\ba_w}[k] + \varepsilon_w[k] \\
\ba_w[k+1]       &= \ba_w[k] + b[k]\bigl(1-\ba_w[k]\bigr)^{2}\,\dW[k] \\
b[k+1]           &= b[k]
\end{align*}

\begin{align*}
F_{dw}[k]        &= \bar f_{dw}[k] + (1-\lambda_c)\sum_{i=1}^{d}\bar f_{dw}[k-i] \\
\varepsilon_w[k] &= \ba_w[k]\,\bar f_T[k] + (1-\lambda_c)\sum_{i=1}^{d}\ba_w[k-i]\,\bar f_T[k-i] - (1-\lambda_c)\,n_w[k]
\end{align*}

\section{Estimator}

\begin{align*}
\mathbf{x} &= \bigl[\,\delta\bw_1,\ \delta\bw_2,\ \delta\bw_3,\ \ba_w,\ b\,\bigr]^{\top} \qquad (d = 2)
\end{align*}

\begin{align*}
\delta\hat{\bw}_1[k+1] &= \delta\hat{\bw}_2[k] \\
\delta\hat{\bw}_2[k+1] &= \delta\hat{\bw}_3[k] \\
\delta\hat{\bw}_3[k+1] &= \lambda_c\,\delta\hat{\bw}_3[k] \\
\dWh[k]                &= \Delta\bw_d[k] + (1-\lambda_c)\,\delta\hat{\bw}_3[k] \\
\ah[k+1]               &= \ah[k] + \bh[k]\bigl(1-\ah[k]\bigr)^{2}\,\dWh[k] \\
\bh[k+1]               &= \bh[k]
\end{align*}

\section{Error dynamics}

(True $-$ Estimator; coefficients at the estimate)

\begin{align*}
\eD[k] &= \dW[k] - \dWh[k] = (1-\lambda_c)\,e_{\delta\bw_3}[k] + F_{dw}[k]\,e_{\ba_w}[k] + \varepsilon_w[k] \\
b[k]\bigl(1-\ba_w[k]\bigr)^{2} &= \bh[k]\bigl(1-\ah[k]\bigr)^{2} - 2\,\bh[k]\bigl(1-\ah[k]\bigr)\,e_{\ba_w}[k] + \bigl(1-\ah[k]\bigr)^{2}e_b[k] + \mathcal{O}\bigl(e^{2}\bigr)
\end{align*}

\begin{align*}
e_{\delta\bw_1}[k+1] &= e_{\delta\bw_2}[k] \\
e_{\delta\bw_2}[k+1] &= e_{\delta\bw_3}[k] \\
e_{\delta\bw_3}[k+1] &= \lambda_c\,e_{\delta\bw_3}[k] - F_{dw}[k]\,e_{\ba_w}[k] - \varepsilon_w[k] \\
e_{\ba_w}[k+1] &= \Bigl[1 + \bh\bigl(1-\ah\bigr)^{2}F_{dw} - 2\bh\bigl(1-\ah\bigr)\bigl(\Delta\bw_d + (1-\lambda_c)\,\delta\hat{\bw}_3\bigr)\Bigr]e_{\ba_w}[k] \\
  &\quad + (1-\lambda_c)\,\bh\bigl(1-\ah\bigr)^{2}e_{\delta\bw_3}[k]
        + \bigl(1-\ah\bigr)^{2}\bigl(\Delta\bw_d + (1-\lambda_c)\,\delta\hat{\bw}_3\bigr)e_b[k]
        + \bh\bigl(1-\ah\bigr)^{2}\varepsilon_w[k] \\
e_b[k+1] &= e_b[k]
\end{align*}

\begin{align*}
F_e[k] = \begin{bmatrix}
0 & 1 & 0 & 0 & 0 \\[2pt]
0 & 0 & 1 & 0 & 0 \\[2pt]
0 & 0 & \lambda_c & -F_{dw} & 0 \\[2pt]
0 & 0 & (1-\lambda_c)\,\bh\bigl(1-\ah\bigr)^{2} &
  \begin{array}{l} 1 + \bh\bigl(1-\ah\bigr)^{2}F_{dw} \\[2pt] \quad - 2\bh\bigl(1-\ah\bigr)\bigl(\Delta\bw_d + (1-\lambda_c)\,\delta\hat{\bw}_3\bigr) \end{array} &
  \begin{array}{l} \bigl(1-\ah\bigr)^{2} \\[2pt] \quad \times\bigl(\Delta\bw_d + (1-\lambda_c)\,\delta\hat{\bw}_3\bigr) \end{array} \\[14pt]
0 & 0 & 0 & 0 & 1
\end{bmatrix}
\end{align*}

\begin{align*}
\mathbf{q}[k] &= \Bigl[\,0,\ 0,\ -\varepsilon_w,\ \bh\bigl(1-\ah\bigr)^{2}\varepsilon_w,\ 0\,\Bigr]^{\top}
\end{align*}


\section{Measurement}

$\bigl(\nabla_d\bw_d[k] := \bw_d[k] - \bw_d[k-d]\bigr)$

\begin{align*}
y_1[k]     &= \delta\bw_1[k] + n_w[k] \\
y_2[k]     &= \ba_w[k-d] + n_a[k], \qquad
\ba_w[k-d]  = \ba_w[k] - b\bigl(1-\ba_w[k]\bigr)^{2}\,\nabla_d\bw_d[k] \\
\hat y_1[k] &= \delta\hat{\bw}_1[k] \\
\hat y_2[k] &= \ah[k] - \bh[k]\bigl(1-\ah[k]\bigr)^{2}\,\nabla_d\bw_d[k] \\
H_{24}[k]  &= \frac{\partial \hat y_2[k]}{\partial \ah[k]} = 1 + 2\,\bh[k]\bigl(1-\ah[k]\bigr)\,\nabla_d\bw_d[k] \\
H_{25}[k]  &= \frac{\partial \hat y_2[k]}{\partial \bh[k]} = -\bigl(1-\ah[k]\bigr)^{2}\,\nabla_d\bw_d[k] \\
H[k]       &= \begin{bmatrix} 1 & 0 & 0 & 0 & 0 \\ 0 & 0 & 0 & H_{24}[k] & H_{25}[k] \end{bmatrix} \\
\mathbf{e}_y[k] &= H[k]\,\mathbf{e}^{-}[k] + \begin{bmatrix} n_w[k] \\ n_a[k] \end{bmatrix}
\end{align*}

\section{Noise}

\begin{align*}
Q[k]    &= Q_{33}[k]\,\mathbf{g}[k]\,\mathbf{g}[k]^{\top}, \qquad
\mathbf{g}[k] = \Bigl[\,0,\ 0,\ -1,\ \bh[k]\bigl(1-\ah[k]\bigr)^{2},\ 0\,\Bigr]^{\top}, \qquad Q_{55} = 0 \\
Q_{33}[k] &= \kappa_T\Bigl(\ah[k] + (1-\lambda_c)^{2}\bigl(\ah[k-1] + \ah[k-2]\bigr)\Bigr) + (1-\lambda_c)^{2}\sigma^{2}_{n_w},

\section{Observability}

($\mathbf{e}[k] = [\,e_{\delta\bw_1},\ e_{\delta\bw_2},\ e_{\delta\bw_3},\ e_{\ba_w},\ e_b\,]^{\top}$,
$\mathbf{n}[k] = [\,n_w[k],\ n_a[k]\,]^{\top}$ \\
$\Phi(k,k_0) := F_e[k-1]\,F_e[k-2]\cdots F_e[k_0]$, $\Phi(k_0,k_0) = I$ \\
$N$ window length \\
$\mathbf{Y} := [\,\mathbf{e}_y[k_0];\ \mathbf{e}_y[k_0+1];\ \dots;\ \mathbf{e}_y[k_0+N]\,]$,
$\mathbf{N} := [\,\mathbf{n}[k_0];\ \mathbf{n}[k_0+1];\ \dots;\ \mathbf{n}[k_0+N]\,]$ \\
$X_{:,j}$ column $j$ of $X$)

\begin{align*}
F_e[k]_{:,5} &= \Bigl[\,0,\ 0,\ 0,\ \bigl(1-\ah\bigr)^{2}\bigl(\Delta\bw_d + (1-\lambda_c)\,\delta\hat{\bw}_3\bigr),\ 1\,\Bigr]^{\top} \\
H[k]_{:,5}   &= \Bigl[\,0,\ -\bigl(1-\ah\bigr)^{2}\,\nabla_d\bw_d\,\Bigr]^{\top}
\end{align*}

\begin{align*}
\mathbf{e}[k]   &= \Phi(k,k_0)\,\mathbf{e}[k_0] \\
\mathbf{e}_y[k] &= H[k]\,\Phi(k,k_0)\,\mathbf{e}[k_0] + \mathbf{n}[k] \\
\mathbf{Y}      &= O(k_0,N)\,\mathbf{e}[k_0] + \mathbf{N}, \qquad
O(k_0,N) := \begin{bmatrix} H[k_0]\,\Phi(k_0,k_0) \\ H[k_0+1]\,\Phi(k_0+1,k_0) \\ \vdots \\ H[k_0+N]\,\Phi(k_0+N,k_0) \end{bmatrix}
\end{align*}

\begin{align*}
R_w &:= \mathrm{diag}\bigl(R[k_0],\ R[k_0+1],\ \dots,\ R[k_0+N]\bigr), \qquad
O_w(k_0,N) := R_w^{-1/2}\,O(k_0,N) \\
\mathbf{Y}_w &:= R_w^{-1/2}\,\mathbf{Y} = O_w(k_0,N)\,\mathbf{e}[k_0] + \mathbf{N}_w, \qquad
\E\bigl[\mathbf{N}_w\mathbf{N}_w^{\top}\bigr] = I
\end{align*}

\begin{align*}
O_w(k_0,N)\,\mathbf{v}_i &= \sigma_i\,\mathbf{u}_i, \qquad i = 1,\dots,5, \qquad
\sigma_1 \ge \dots \ge \sigma_5 \ge 0, \qquad \|\mathbf{v}_i\| = \|\mathbf{u}_i\| = 1 \\
\sigma_1 &= \max_{\|\mathbf{v}\|=1}\|O_w(k_0,N)\,\mathbf{v}\|, \qquad
\sigma_5  = \min_{\|\mathbf{v}\|=1}\|O_w(k_0,N)\,\mathbf{v}\| \\
\operatorname{rank} O_w(k_0,N) &= \#\{\,i : \sigma_i > 0\,\} \\
F_e[k]_{:,j} = I_{:,j},\ \ H[k]_{:,j} = 0\ \ \forall k
  &\ \Rightarrow\ O_w(k_0,N)_{:,j} = 0 \ \Rightarrow\ \sigma_5 = 0
\end{align*}

\begin{align*}
\hat{\mathbf{e}}[k_0] &= \sum_{i=1}^{5}\frac{\mathbf{u}_i^{\top}\mathbf{Y}_w}{\sigma_i}\,\mathbf{v}_i \\
\mathbf{v}_i^{\top}\bigl(\hat{\mathbf{e}}[k_0] - \mathbf{e}[k_0]\bigr)
  &= \frac{\mathbf{u}_i^{\top}\mathbf{N}_w}{\sigma_i}, \qquad
\E\Bigl[\Bigl(\mathbf{v}_i^{\top}\bigl(\hat{\mathbf{e}}[k_0] - \mathbf{e}[k_0]\bigr)\Bigr)^{2}\Bigr] = \frac{1}{\sigma_i^{2}} \\
\mathbf{v}_5 \to \bigl[\,0,\ 0,\ 0,\ 0,\ 1\,\bigr]^{\top}
  &\ \Rightarrow\ \E\bigl[(\hat{e}_b - e_b)^{2}\bigr] \to \frac{1}{\sigma_5^{2}} \\
\frac{1}{\sigma_5} &\lessgtr \sqrt{P_{55}[0]}, \qquad
\sqrt{P_{55}[0]} = \sup_{\bw\in\text{env}}\Bigl|\,b_{\mathrm{true}}(\bw) - \tfrac{8}{9}\,\Bigr|
\end{align*}
