\section{Gain law}

(bars denote normalized quantities: lengths by $R$; $a_o := \Delta t/(\gamma_N R)$ [1/pN] and $a_{\mathrm{nom}} := \Delta t/\gamma_N = a_o R$ [$\mu$m/pN];
$\ba := a/a_{\mathrm{nom}}$; forces by $a_o$: $\bar f := a_o f$, so that $\ba\,\bar f = \dW$; $0 \le \ba \le 1$, far field 1, contact 0;
hats sit on the normalized quantities $\ah$, $\bh$, $\delta\hat{\bw}$; $e_\theta = \theta - \hat\theta$)

\begin{align*}
\ba_{\mathrm{true}}(\bw)  &= 1 - \frac{9}{8\bw} + \frac{1}{2\bw^{3}} - \frac{57}{100\bw^{4}} + \frac{1}{5\bw^{5}} + \frac{7}{200\bw^{11}} - \frac{1}{25\bw^{12}} \\
\ap_{\mathrm{true}}(\bw)  &= \frac{d\ba_{\mathrm{true}}}{d\bw} \\
\app_{\mathrm{true}}(\bw) &= \frac{d^{2}\ba_{\mathrm{true}}}{d\bw^{2}}
\end{align*}

\begin{align*}
\ap_{\mathrm{true}}(\bw) > 0 \ \text{on}\ \bw > 1 \quad &\Longrightarrow \quad
  \ap = b_{\mathrm{true}}(\ba)\bigl(1-\ba\bigr)^{2}\ \text{exactly}, \qquad
  b_{\mathrm{true}} := \frac{\ap_{\mathrm{true}}}{\bigl(1-\ba_{\mathrm{true}}\bigr)^{2}} \\
\ba_{\mathrm{true}}(1) = 0,\ \ap_{\mathrm{true}}(1) = 1 \quad &\Longrightarrow \quad b_{\mathrm{true}}\bigl(\ba = 0\bigr) = 1 \\
1 - \ba_{\mathrm{true}} = \frac{9}{8\bw} + \mathcal{O}(\bw^{-3}) \quad &\Longrightarrow \quad b_{\mathrm{true}}\bigl(\ba \to 1\bigr) = \frac{8}{9} \\
\app_{\mathrm{true}}(\bw) &= b'_{\mathrm{true}}(\bw)\bigl(1-\ba_{\mathrm{true}}\bigr)^{2} - 2\,b_{\mathrm{true}}\bigl(1-\ba_{\mathrm{true}}\bigr)\,\ap_{\mathrm{true}}
\end{align*}

\begin{align*}
\ba(\bw)         &= 1 - \frac{1}{b\,(\bw - \bw_s)} \\
\ap(\ba, b)      &:= b\bigl(1-\ba\bigr)^{2} \\
\app(\ba, b)     &:= \frac{d\ap}{d\bw} = \frac{\partial\ap}{\partial\ba}\,\ap = -2\,b\bigl(1-\ba\bigr)\ap = -2\,b^{2}\bigl(1-\ba\bigr)^{3} \\
a_w &= a_o\,\ba_w , \qquad a_o = \frac{\Delta t}{\gamma_N R}
\end{align*}

\section{Error dynamics}

\begin{align*}

\end{align*}

\begin{align*}

   + \underbrace{\ap\,\eD + A\,\dWh\,e_{\ba_w} + \dWh\,f}_{(1)}
