\section{Regulation of the motion gain}

\hd{Gain step of the probe (Section 8)}
\[
\ba[k+1] - \ba[k] = \ba'[k]\,\big(\Delta\bx_d[k] + (1-\lc)\,\delta\bx_3[k] + F_{dx}[k]\,e_{\ba}[k] + \varepsilon_x[k]\big)
\]

\hd{Regulation objective and law}
\[
\begin{aligned}
\ba_d - \ba[k+1] &= \lambda_a\,\big(\ba_d - \ba[k]\big) \\
\Delta\bx_d[k] &= (1-\lambda_a)\,\frac{\ba_d - \hat{\ba}[k]}{\hat{\ba}'[k]} \\
\bx_d[k+1] &= \bx_d[k] + \Delta\bx_d[k] \\
\lambda_a &= 1 - T_s/\tau_a, \qquad \tau_a = 0.1\,\mathrm{s}
\end{aligned}
\]
