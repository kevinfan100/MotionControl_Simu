\section{Law: slope only}

($h := \bw - \bw_s$, $\bw_s$ = visible surface, given; $\gs := h + \dl$ the shifted gap; $\dl > 0$ = effective wall below the visible surface;
contact slope pinned: $\ap(\gs = 1) = 1$ for every $p$; $\dl$, $p$ = the two constants; used on $h \ge 1.1$)

\begin{align*}
\ap(h;\,\dl,\,p)                        &= \frac{1}{(h + \dl)^{p}} \\[2pt]
\frac{\partial\ap}{\partial h}          &= -\,\frac{p\,\ap}{h + \dl} \\[2pt]
\frac{\partial\ap}{\partial\dl}         &= -\,\frac{p\,\ap}{h + \dl} \qquad = \frac{\partial\ap}{\partial h} \\[2pt]
\frac{\partial\ap}{\partial p}          &= -\,\ap\,\ln(h + \dl) \\[2pt]
\frac{\partial\ap}{\partial\ba_w}       &= 0
\end{align*}

(integrated once, for the seed container only, $\ba(\infty) = 1$; the value law is NOT used by the estimator)
\begin{align*}
\int_{h}^{\infty}\ap\,dh'               &= \frac{1}{(p - 1)(h + \dl)^{p-1}} \\
\ba(h;\,\dl,\,p)                        &= 1 - \frac{1}{(p - 1)(h + \dl)^{p-1}} \\
\frac{\partial\ba}{\partial\dl}         &= \frac{1}{(h + \dl)^{p}} = \ap \\
\frac{\partial\ba}{\partial p}          &= \frac{\dfrac{1}{p-1} + \ln(h + \dl)}{(p - 1)(h + \dl)^{p-1}}
\end{align*}

(family residual on a wall with constants $(\dl_w, p_w)$, and the gain error it leaves along a descent from $h_0$ if never corrected)
\begin{align*}
r'(h)   &:= \ap_{\mathrm{true}}(h) - \ap(h;\,\dl_w,\,p_w) \\
E(h)    &:= \int_{h}^{h_0} r'(h')\,dh'
\end{align*}
