\section{Gain law (slope only)}

(bars denote normalized quantities: lengths by $R$; $a_o := \Delta t/(\gamma_N R)$ [1/pN]; $\ba := a/a_{\mathrm{nom}}$;
$\bar f := a_o f$, so that $\ba\,\bar f = \dW$; $\bw$ = particle centre along the wall normal; $\bw_s$ = the visible surface
on the same axis, given (plane: the wall, contact at $h = 1$; cell: the visible top); $h := \bw - \bw_s$;
$p$ = fixed exponent; $c$ = the one constant of the wall; the law is used on $h \ge 1.1$; $e_\theta = \theta - \hat\theta$)

\begin{align*}
\ap(h;\,c)                              &:= \frac{1 - c}{h^{p}} \\[4pt]
\frac{\partial\ap}{\partial c}          &= -\,\frac{1}{h^{p}} = -\,\frac{\ap}{1 - c} \\[4pt]
\frac{\partial\ap}{\partial h}          &= -\,\frac{p\,\ap}{h} \\[4pt]
\frac{\partial\ap}{\partial\ba_w}       &= 0 \\[4pt]
\int_{h}^{\infty}\ap(s;\,c)\,ds         &= \frac{1 - c}{(p-1)\,h^{p-1}}, \qquad
p = 2,\ c = 0:\quad \int_{1}^{\infty}\ap\,ds = 1 \quad(\ba(1) = 0,\ \ba(\infty) = 1)
\end{align*}

(no value law is written: $\ba_w$ is a state; the law only gives its increment per step of height)

\section{True dynamics}

$\bigl(\delta\bw_j[k] := \delta\bw[k-d+j-1],\ j = 1,\dots,d+1,\ d = 2$; $\ap_{\mathrm{true}}$ = the plant's slope, not in the family$\bigr)$

\begin{align*}
\bw[k]           &= \bw_d[k] - \delta\bw_3[k] \\
\bw_d[k+1]       &= \bw_d[k] + \Delta\bw_d[k] \\
\ba_w[k+1]       &= \ba_w[k] + \ap_{\mathrm{true}}\bigl(\bw[k] - \bw_s\bigr)\,\dW[k] + O(\dW^{2}) \\
\delta\bw_1[k+1] &= \delta\bw_2[k] \\
\delta\bw_2[k+1] &= \delta\bw_3[k] \\
\delta\bw_3[k+1] &= \lambda_c\,\delta\bw_3[k] - F_{dw}[k]\,e_{\ba_w}[k] - \varepsilon_w[k] \\
\dW[k]           &= \Delta\bw_d[k] + (1-\lambda_c)\,\delta\bw_3[k] + F_{dw}[k]\,e_{\ba_w}[k] + \varepsilon_w[k] \\
c[k+1]           &= c[k]
\end{align*}

\begin{align*}
F_{dw}[k]        &= \bar f_{dw}[k] + (1-\lambda_c)\sum_{i=1}^{d}\bar f_{dw}[k-i] \\
\varepsilon_w[k] &= \ba_w[k]\,\bar f_T[k] + (1-\lambda_c)\sum_{i=1}^{d}\ba_w[k-i]\,\bar f_T[k-i] - (1-\lambda_c)\,n_w[k] \\
r'[k]            &:= \ap_{\mathrm{true}}\bigl(\bw[k] - \bw_s\bigr) - \ap\bigl(\bw[k] - \bw_s;\,c\bigr) \\
                 &\qquad\text{(slope residual of the family on this wall; zero only if the wall is in the family)}
\end{align*}
