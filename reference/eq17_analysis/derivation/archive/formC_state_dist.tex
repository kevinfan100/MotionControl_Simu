\section{Gain law}

(bars denote normalized quantities: lengths by $R$, $\ba = a/a_o$,
$\bar{f} = a_o f$; $0 < \ba < 1$; $e_{\theta} = \theta - \hat{\theta}$)

\begin{equation*}
\ba_{\mathrm{true}}(\bw) = 1 - \frac{9}{8\bw} + \frac{1}{2\bw^{3}}
  - \frac{57}{100\,\bw^{4}} + \frac{1}{5\bw^{5}}
  + \frac{7}{200\,\bw^{11}} - \frac{1}{25\,\bw^{12}}
\end{equation*}

\begin{equation*}
\ba'_{\mathrm{true}}(\bw) > 0 \ \ \text{on} \ \ \bw > 1
\qquad\Longrightarrow\qquad
\ba' = \frac{\bigl(1-\ba\bigr)^{2}}{b_{\mathrm{true}}(\ba)}
\ \ \text{exactly},
\qquad
b_{\mathrm{true}} := \frac{\bigl(1-\ba_{\mathrm{true}}\bigr)^{2}}
  {\ba'_{\mathrm{true}}}
\end{equation*}

\begin{align*}
\ba_{\mathrm{true}}(1) = 0, \quad \ba'_{\mathrm{true}}(1) = 1
  &\quad\Longrightarrow\quad b_{\mathrm{true}}\bigl(\ba = 0\bigr) = 1 \\
1 - \ba_{\mathrm{true}} = \frac{9}{8\bw} + \mathcal{O}(\bw^{-3})
  &\quad\Longrightarrow\quad b_{\mathrm{true}}\bigl(\ba \to 1\bigr) = \frac{9}{8}
\end{align*}

\begin{align*}
\ba'(\ba) &:= \frac{\mathrm{d}\ba}{\mathrm{d}\bw} = \bigl(1-\ba\bigr)^{2}
  \qquad \bigl(b_{\mathrm{true}}(\ba) \to 1, \ \text{its contact value}\bigr) \\
a_w &= a_o\,\ba_w , \qquad a_o = \frac{\Delta t}{\gamma_N R}
\end{align*}

\begin{equation*}
\frac{\ba'(\ba_{\mathrm{true}})}{\ba'_{\mathrm{true}}}
  = b_{\mathrm{true}}
  = \frac{9}{8} + \frac{1}{2\bw^{2}} - \frac{57}{50\,\bw^{3}}
  + \mathcal{O}(\bw^{-4})
\end{equation*}

\section{True dynamics}

$\bigl(\delta\bw_j[k] := \delta\bw[k-d+j-1],\ j = 1,\dots,d+1,\ d = 2\bigr)$

\begin{align*}
\bw[k]           &= \bw_d[k] - \delta\bw_3[k] \\
\bw_d[k+1]       &= \bw_d[k] + \Delta\bw_d[k] \\
\delta\bw_1[k+1] &= \delta\bw_2[k] \\
\delta\bw_2[k+1] &= \delta\bw_3[k] \\
\delta\bw_3[k+1] &= \lambda_c\,\delta\bw_3[k] - F_{dw}[k]\,e_{\ba_w}[k]
  - \varepsilon_w[k]
\end{align*}

\begin{align*}
F_{dw}[k] &= \bar{f}_{dw}[k] + (1-\lambda_c)\sum_{i=1}^{d}\bar{f}_{dw}[k-i] \\
\varepsilon_w[k] &= \ba_w[k]\,\bar{f}_T[k]
  + (1-\lambda_c)\sum_{i=1}^{d}\ba_w[k-i]\,\bar{f}_T[k-i]
  + (1-\lambda_c)\,n_w[k]
\end{align*}

\begin{equation*}
\ba_w[k+1] = \ba_w[k] + \ba'_{\mathrm{true}}[k]\,\Bigl[
  \Delta\bw_d[k] + (1-\lambda_c)\,\delta\bw_3[k]
  + F_{dw}[k]\,e_{\ba_w}[k] + \varepsilon_w[k]\Bigr]
\end{equation*}

\subsection*{(a)}

\begin{equation*}
\ba_w[k+1] = \ba_w[k] + \bigl(1-\ba_w[k]\bigr)^{2}\Bigl[
  \Delta\bw_d[k] + (1-\lambda_c)\,\delta\bw_3[k]
  + F_{dw}[k]\,e_{\ba_w}[k] + \varepsilon_w[k]\Bigr]
\end{equation*}

\subsection*{(b)}

\begin{equation*}
\ba_w[k+1] = \ba_w[k] + \bigl(1-\ba_w[k]\bigr)^{2}\Bigl[
  \Delta\bw_d[k] + (1-\lambda_c)\,\delta\bw_3[k]
  + F_{dw}[k]\,e_{\ba_w}[k] + \varepsilon_w[k]\Bigr] + \delta a[k]
\end{equation*}

\begin{equation*}
\delta a[k]
  = \Bigl[\ba'_{\mathrm{true}}[k] - \bigl(1-\ba_w[k]\bigr)^{2}\Bigr]

\begin{equation*}

\end{equation*}
