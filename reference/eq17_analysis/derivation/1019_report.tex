\section{Regulation of the motion gain}

\hd{Regulation law}
\[
\begin{aligned}
\Delta\bx_d[k] &= (1-\lambda_a)\,\frac{\ba_d - \hat{\ba}[k]}{\hat{\ba}'[k]} \\
\bx_d[k+1] &= \bx_d[k] + \Delta\bx_d[k] \\
\lambda_a &= 1 - T_s/\tau_a, \qquad \tau_a = 0.1\,\mathrm{s}
\end{aligned}
\]

\hd{$\ba'$ as a state variable: limit on $\Delta\bx_d$}
\[
\begin{aligned}
\big|\Delta\bx_d[k]\big| &\le v_{\max} \\
v_{\max} &= (22.2 - 1.10)\,T_s / 1\,\mathrm{s}
\end{aligned}
\]
