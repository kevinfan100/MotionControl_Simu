\section{Gain law}

($e_\theta = \theta - \hat\theta$; coefficients at the estimate)

\begin{align*}
\ba_{\mathrm{true}}(\bw) &= \frac{1}{c_\perp(\bw)} \\
u(\bw) &:= \frac{1}{1 - \ba_{\mathrm{true}}(\bw)} \\
b_{\mathrm{true}}(\bw) &:= \frac{\mathrm{d}u}{\mathrm{d}\bw} = \frac{\ba_{\mathrm{true}}'(\bw)}{\bigl(1-\ba_{\mathrm{true}}(\bw)\bigr)^{2}} \\
\ba_{\mathrm{true}}'(\bw) &= b_{\mathrm{true}}(\bw)\bigl(1-\ba_{\mathrm{true}}(\bw)\bigr)^{2} \\
\lim_{\bw \to \infty} b_{\mathrm{true}} &= 1, \qquad \lim_{\bw \to 1} b_{\mathrm{true}} = \frac{8}{9}
\end{align*}

\section{Model}

\begin{align*}
b_{\mathrm{true}}(\bw) &\longrightarrow \bh \quad \text{(constant)} \\
\Delta b(\bw) &:= b_{\mathrm{true}}(\bw) - \bh
\end{align*}

\section{Step}

\begin{align*}
\frac{1}{1-\ah[k+1]} &= \frac{1}{1-\ah[k]} + \bh[k]\,\dWh[k] \\
\frac{1}{1-\ba_{\mathrm{true}}[k+1]} &= \frac{1}{1-\ba_{\mathrm{true}}[k]}
   + \int_{\bw[k]}^{\bw[k]+\Delta\bw[k]} b_{\mathrm{true}}(\bw)\,\mathrm{d}\bw
\end{align*}

\section{Law error of one step}

\begin{align*}
\frac{\partial\,\ah[k+1]}{\partial\,\bh[k]} &= \bigl(1-\ah[k+1]\bigr)^{2}\,\dWh[k]
   = \Bigl(\frac{1-\ah[k+1]}{1-\ah[k]}\Bigr)^{2}\bigl(1-\ah[k]\bigr)^{2}\,\dWh[k] \\[4pt]
\varepsilon_{\mathrm{law}}[k] &= \bigl(1-\ah[k]\bigr)^{2}\,\Delta b\bigl(\bw[k]\bigr)\,\dWh[k] + \mathcal{O}\bigl(\dWh^{2}\bigr)
\end{align*}
