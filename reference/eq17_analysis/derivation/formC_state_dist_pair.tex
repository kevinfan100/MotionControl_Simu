\section{Gain law}

(bars denote normalized quantities: lengths by $R$, $\ba = a/a_o$,
$\bar{f} = a_o f$; $0 < \ba < 1$; $e_{\theta} = \theta - \hat{\theta}$;
$b$ is a FIXED constant here, not a state)

\begin{equation*}
\ba_{\mathrm{true}}(\bw) = 1 - \frac{9}{8\bw} + \frac{1}{2\bw^{3}}
  - \frac{57}{100\,\bw^{4}} + \frac{1}{5\bw^{5}}
  + \frac{7}{200\,\bw^{11}} - \frac{1}{25\,\bw^{12}}
\end{equation*}

\begin{equation*}
\ba'_{\mathrm{true}}(\bw) > 0 \ \ \text{on} \ \ \bw > 1
\qquad\Longrightarrow\qquad
\ba' = b_{\mathrm{true}}(\ba)\bigl(1-\ba\bigr)^{2}
\ \ \text{exactly},
\qquad
b_{\mathrm{true}} := \frac{\ba'_{\mathrm{true}}}
  {\bigl(1-\ba_{\mathrm{true}}\bigr)^{2}}
\end{equation*}

\begin{align*}
\ba_{\mathrm{true}}(1) = 0, \quad \ba'_{\mathrm{true}}(1) = 1
  &\quad\Longrightarrow\quad b_{\mathrm{true}}\bigl(\ba = 0\bigr) = 1 \\
1 - \ba_{\mathrm{true}} = \frac{9}{8\bw} + \mathcal{O}(\bw^{-3})
  &\quad\Longrightarrow\quad b_{\mathrm{true}}\bigl(\ba \to 1\bigr) = \frac{8}{9}
\end{align*}

\begin{align*}
\ba'(\ba) &:= b\bigl(1-\ba\bigr)^{2} , \qquad b = \tfrac{8}{9} \ \text{fixed} \\
a_w &= a_o\,\ba_w , \qquad a_o = \frac{\Delta t}{\gamma_N R}
\end{align*}

\section*{Observability}

Gate 1. Both routes for $\bwD$:

\begin{align*}
\text{state route} \quad
  \frac{\partial \delta\bw_3[k+1]}{\partial \bwD} &= -1 ,
  \qquad
  \frac{\partial \ba_w[k+1]}{\partial \bwD} = b\bigl(1-\hat{\ba}_w\bigr)^{2} \\
\text{measurement route} \quad
  \frac{\partial y_1}{\partial \bwD} = \frac{\partial y_2}{\partial \bwD} &= 0
\end{align*}

Gate 2. Time signatures on the position channel:

\begin{align*}
e_{\ba_w} &\ \longrightarrow\ -F_{dw}[k]\,e_{\ba_w}
  \qquad \text{scaled by the applied force} \\
e_{\bwD}  &\ \longrightarrow\ -e_{\bwD}
  \qquad\qquad\ \ \text{unit coefficient} \\
e_{\delta\bwD} &\ \longrightarrow\ \text{ramp}
\end{align*}

Gate 3. Two reductions. Through $y_2$, with $\bwD$ constant and
$h := 1 + 2b(1-\hat{\ba}_w)\nabla_d\bw_d$:

\begin{align*}
A_2 &= \begin{bmatrix}
  1 + b\bigl(1-\hat{\ba}_w\bigr)^{2}F_{dw}
    - 2b\bigl(1-\hat{\ba}_w\bigr)
      \bigl(\Delta\bw_d + (1-\lambda_c)\delta\hat{\bw}_3\bigr)
  & b\bigl(1-\hat{\ba}_w\bigr)^{2} \\[0.4em]
  0 & 1
\end{bmatrix} ,
\qquad
C_2 = \begin{bmatrix} h & 0 \end{bmatrix} \\[0.6em]
\det O_2 &= \det\begin{bmatrix} C_2 \\ C_2 A_2 \end{bmatrix}
  = h^{2}\,b\bigl(1-\hat{\ba}_w\bigr)^{2} = h^{2}\,\ba' \\[0.4em]
\det O_2 = 0 &\iff \ba' = 0 \ \ (\ba \to 1, \ \text{far field})
  \quad\text{or}\quad \nabla_d\bw_d = -\frac{1}{2b(1-\hat{\ba}_w)}
\end{align*}

Through $y_1$, the channel sees only the combination
$F_{dw}[k]\,e_{\ba_w} + e_{\bwD}$, so two consecutive steps give

\begin{align*}
\det\begin{bmatrix} F_{dw}[k] & 1 \\ F_{dw}[k+1] & 1 \end{bmatrix}
  &= F_{dw}[k] - F_{dw}[k+1] \\
\det O_2 = 0 &\iff F_{dw}[k+1] = F_{dw}[k]
  \quad \text{(constant applied force)}
\end{align*}
