\end{align*}

\section{Error of the surface normal from the error of $\Ah$}
\setline{$\mathbf e_a = \av - \avh = [\,e_{xx}\ e_{yy}\ e_{zz}\ e_{xy}\ e_{xz}\ e_{yz}\,]^{\mathsf T}$; first order}
\begin{align*}
e_{\theta_y} &= \frac{\big(\Ah_{xx} - \Ah_{zz}\big)\,(-2\,e_{xz}) + 2\,\Ah_{xz}\,\big(e_{xx} - e_{zz}\big)}{2\,\Big(\big(\Ah_{xx} - \Ah_{zz}\big)^2 + 4\,\Ah_{xz}^2\Big)} \\
e_{\varphi_{yz}} &= \frac{\big(\Ah_{yy} - \Ah_{zz}\big)\,(-2\,e_{yz}) + 2\,\Ah_{yz}\,\big(e_{yy} - e_{zz}\big)}{2\,\Big(\big(\Ah_{yy} - \Ah_{zz}\big)^2 + 4\,\Ah_{yz}^2\Big)} \\
e_{\theta_x} &= -\frac{\cos\hat\theta_y\,\big(1 + \tan^2\hat\varphi_{yz}\big)}{1 + \tan^2\hat\varphi_{yz}\cos^2\hat\theta_y}\,e_{\varphi_{yz}}
+ \frac{\tan\hat\varphi_{yz}\,\sin\hat\theta_y}{1 + \tan^2\hat\varphi_{yz}\cos^2\hat\theta_y}\,e_{\theta_y} \\
\ww - \wh &= -\vh\,e_{\theta_x} + \cos\hat\theta_x\,\uh\,e_{\theta_y}
\end{align*}
