\begin{center}\textbf{\large c-only slope law under the journal control law (J6)}\end{center}

($h := \bw - \bw_s$, $\bw_s$ = visible surface, given ($\bw_s = 0$ in every c-only run); $p = 2$ pinned; $c$ = the one constant;
$d = 2$; $e := \text{true} - \text{estimate}$; all lengths $/R$, gain $/a_{nom}$, force $\times a_o$)

\section{Law and control law}

(gain law: unchanged from 0928 / 1006)
\begin{align*}
\ap(h;\,c)                        &= \frac{1 - c}{h^{2}} \\[2pt]
\frac{\partial\ap}{\partial h}    &= -\,\frac{2\,\ap}{h} \\[2pt]
\frac{\partial\ap}{\partial c}    &= -\,\frac{1}{h^{2}} \\[2pt]
\frac{\partial\ap}{\partial\ba_w} &= 0
\end{align*}

(control law; every estimate on the right is the posterior of step $k$, Section 4)
\begin{align*}
\fd[k]    &= \frac{1}{\ah[k]}\Bigl(\dW_d[k] + (1-\lc)\,\delta\hat{\bw}_3[k] - \hat{x}_D[k]\Bigr), \qquad \hat{x}_D \equiv 0 \J{6} \\[4pt]
M[k]      &:= \dW_d[k] + (1-\lc)\,\delta\hat{\bw}_3[k] \qquad = \ah[k]\,\fd[k] \\[2pt]
\fdet[k]  &:= \frac{\dW_d[k]}{\ah[k]} \qquad(\text{the exogenous share of } \fd;\ \text{regressor of } F_e)
\end{align*}

(reference: the law this file replaces, \texttt{motion\_control\_law\_formC\_b.m:1773}; $\nabla_d\bw_d[k] := \bw_d[k] - \bw_d[k-d]$)
\begin{align*}
\fd^{\mathrm{old}}[k] &= \frac{1}{\ah[k-1]}\Bigl(\dW_d[k] + (1-\lc)\Bigl[\delta\bw_m[k] + \nabla_d\bw_d[k] - \sum_{i=1}^{d}\ah[k-i]\,\fd^{\mathrm{old}}[k-i]\Bigr]\Bigr)
\end{align*}

\section{True dynamics}

\begin{align*}
\bw[k+1]         &= \bw[k] + \ba_w[k]\,\fd[k] + w_T[k] \J{5} \\
\delta\bw_3[k]   &= \bw_d[k] - \bw[k] \\
\delta\bw_1[k+1] &= \delta\bw_2[k] \\
\delta\bw_2[k+1] &= \delta\bw_3[k] \\
\delta\bw_3[k+1] &= \delta\bw_3[k] + \dW_d[k] - \ba_w[k]\,\fd[k] - w_T[k] \\
                 &= \lc\,\delta\bw_3[k] + (1-\lc)\,e_{\delta\bw_3}[k] - \fd[k]\,e_{\ba_w}[k] - w_T[k] \J{7, 8, 14} \\

\dW[k]           &:= \bw[k+1] - \bw[k] = M[k] + \fd[k]\,e_{\ba_w}[k] + w_T[k]
\end{align*}

\section{Estimator}

($\hh[k] = \bw_d[k] - \delta\hat{\bw}_3[k] - \bw_s$; $\ap[k] = (1-\ch[k])/\hh[k]^{2}$; superscript $-$ = before the update with $y[k]$)

\begin{align*}

\delta\hat{\bw}_3^{-}[k] &= \lc\,\delta\hat{\bw}_3[k-1] \J{16} \\
