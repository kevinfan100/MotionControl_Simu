\hd{Symbols and units}
\begin{center}
\begin{tabular}{lll}
\toprule
Symbol & Meaning & Unit \\
\midrule
$x[k]$              & position along axis            & $\mu$m \\
$\bar x := x/R$     & normalized position             & -- \\
$a_x[k]$            & motion gain (mobility)          & $\mu$m/pN \\
$a'_x := da_x/dx$   & gain sensitivity                & 1/pN \\
$K_x(\bar x)$       & wall sensitivity of $c$          & -- \\
$f_T[k]$            & thermal force                   & pN \\
$\sigma^2_x[k]$     & single-step thermal position var & $\mu$m$^2$ \\
$a_{det}[k]$        & deterministic part of $a_x$      & $\mu$m/pN \\
$a_{ram}[k]$        & random part of $a_x$             & $\mu$m/pN \\
$x_{ram}[k]$        & position residual                & $\mu$m \\
\bottomrule
\end{tabular}
\end{center}

\medskip
\hd{Wall model}
\[ a_x[k] = \frac{a_{nom}}{c(\bar x[k])} \]

\medskip
\hd{Wall sensitivity}
\[
K_x(\bar x) := \frac{1}{c}\frac{dc}{d\bar x},
\qquad
a'_x[k] := \frac{d a_x}{d x} = -\frac{a_x[k]\,K_x(\bar x[k])}{R}
\]

\medskip
\hd{Thermal force}
\begin{empheq}[box=\fbox]{equation*}
\mathrm{Var}\big(f_T[k]\big) = \frac{4\,k_B T}{a_x[k]}
\end{empheq}

\medskip
\hd{Thermal-driven position variance (single step)}
\begin{empheq}[box=\fbox]{equation*}
\sigma^2_x[k] := a_x[k]^2\cdot \mathrm{Var}\big(f_T[k]\big) = 4\,k_B T\,a_x[k]
\end{empheq}

\medskip
\hd{Gain decomposition}
\[
a_x[k] = a_{det}[k] + a_{ram}[k],
\qquad
a_{det}[m] = \frac{a_{nom}}{c(\bar x_d[m])}
\]
\[
a_{ram}[k] \approx a'_x[k]\cdot x_{ram}[k],
\qquad
x_{ram}[k] := x[k] - x_d[k]
\]

\medskip
\hd{Variance propagation}
\begin{empheq}[box=\fbox]{align*}
\mathrm{Var}\big(a_{ram}[k]\big)   &= a_x'^{\,2}[k]\cdot \mathrm{Var}\big(x_{ram}[k]\big) \\[0.8ex]
\mathrm{Var}\big(\Delta a_{ram}[k]\big) &= a_x'^{\,2}[k]\cdot \mathrm{Var}\big(\Delta x_{ram}[k]\big)
\end{empheq}

\medskip
\hd{Deterministic increment}
\[
\Delta a_x[k] = a_{det}[k+1] - a_{det}[k] = a'_x[k]\cdot \Delta x_d[k]
\]
