\end{center}

\section{Regulation law (normal direction $\bw$; 1019 Part III, flat wall $\bw = z$)}

\hd{Regulation law}
\[
\begin{aligned}
\Delta\bw_d[k] &= \frac{T_s}{\tau_a}\,\frac{\ba_d - \hat{\ba}_N[k]}{\hat{\ba}'_N[k]} \\
\bw_d[k+1] &= \bw_d[k] + \Delta\bw_d[k]
\end{aligned}
\]

\hd{Motion gain under the regulation law ($\hat{\ba}_N = \ba_N$, $\hat{\ba}'_N = \ba'_N$, first order in $\Delta\bw_d$)}
\[
\begin{aligned}
\ba_N[k+1] &= \ba_N[k] + \ba'_N[k]\,\Delta\bw_d[k] \\
\ba_d - \ba_N[k+1] &= \lambda_a\,\big(\ba_d - \ba_N[k]\big) \\
\lambda_a &= 1 - T_s/\tau_a
\end{aligned}
\]

\section{Three axes: regulation along $\hat{\mathbf{w}}$, control of $\bar{u}_d$, $\bar{v}_d$}

\hd{Desired position along the wall in the estimated local coordinate, relative to the start}
\[
\begin{aligned}
\bar{u}_d[k] &= \hat{\mathbf{u}}[k]^{\mathsf{T}}\,\big(\bar{\mathbf{p}}_d[k] - \bar{\mathbf{p}}_d[0]\big) \\
\bar{v}_d[k] &= \hat{\mathbf{v}}[k]^{\mathsf{T}}\,\big(\bar{\mathbf{p}}_d[k] - \bar{\mathbf{p}}_d[0]\big)
\end{aligned}
\]

\hd{Control law}
\[
\begin{aligned}
\Delta\bw_d[k] &= \frac{T_s}{\tau_a}\,\frac{\ba_d - \hat{\ba}_N[k]}{\hat{\ba}'_N[k]} \\
\Delta\bar{u}_d[k] &= -\frac{T_s}{\tau_u}\,\bar{u}_d[k] \\
\Delta\bar{v}_d[k] &= -\frac{T_s}{\tau_u}\,\bar{v}_d[k] \\
\bar{\mathbf{p}}_d[k+1] &= \bar{\mathbf{p}}_d[k] + \hat{\mathbf{V}}[k] \begin{bmatrix} \Delta\bar{u}_d[k] \\ \Delta\bar{v}_d[k] \\ \Delta\bw_d[k] \end{bmatrix}
\end{aligned}
\]

\hd{Under the control law ($\hat{\mathbf{V}}[k+1] = \hat{\mathbf{V}}[k]$)}
\[
\begin{aligned}
\ba_d - \ba_N[k+1] &= \lambda_a\,\big(\ba_d - \ba_N[k]\big) \\
\bar{u}_d[k+1] &= \lambda_u\,\bar{u}_d[k] \\
\bar{v}_d[k+1] &= \lambda_u\,\bar{v}_d[k] \\
\lambda_a &= 1 - T_s/\tau_a \\
\lambda_u &= 1 - T_s/\tau_u
\end{aligned}
\]

\hd{Change of $\hat{\mathbf{V}}$ between two steps (first order in $\Delta\hat{\theta}_y$, $\hat{\theta}_x = 0$)}
\[
\bar{u}_d[k+1] = \lambda_u\,\bar{u}_d[k] - \Delta\hat{\theta}_y[k]\;\hat{\mathbf{w}}[k]^{\mathsf{T}}\big(\bar{\mathbf{p}}_d[k+1] - \bar{\mathbf{p}}_d[0]\big)
\]

\hd{Setting}
\[
\tau_a = 2\,\mathrm{s}, \qquad \sqrt{p_{\theta}[0]} = 10^{\circ}, \qquad q_{\theta} = 0
\]

\section{Hold, approach, scan}

\hd{Angle information per sample (angle rows of the output, random component)}
\[
I_{\theta}[k] \approx \frac{\big(\ba_T[k] - \ba_N[k]\big)^2}{2.5\,\ba_T[k]\,\ba_N[k]}
\]

\hd{Process noise of the angles}
\[
\begin{aligned}
q_{\theta}[k] &= 0, && t < t_{\mathrm{scan}} \\
q_{\theta}[k] &= \Pfix^2\, I[k], && t \ge t_{\mathrm{scan}} \\
I[k] &= 1/p_{\theta}[k] - 1/p^{f}_{\theta}[k]
\end{aligned}
\]

\hd{Desired position}
\[
\begin{aligned}
\Delta\bw_d[k] &= 0, && t < t_{\mathrm{hold}} \\
\Delta\bw_d[k] &= \frac{T_s}{\tau_a}\,\frac{\ba_d - \hat{\ba}_N[k]}{\hat{\ba}'_N[k]}, && t \ge t_{\mathrm{hold}} \\
\Delta\bar{u}_d[k] &= \Delta\bar{u}_{\mathrm{scan}}, && t \ge t_{\mathrm{scan}} \\
\bar{\mathbf{p}}_d[k+1] &= \bar{\mathbf{p}}_d[k] + \hat{\mathbf{V}}[k] \begin{bmatrix} \Delta\bar{u}_d[k] \\ 0 \\ \Delta\bw_d[k] \end{bmatrix}
\end{aligned}
\]

\hd{Setting}
\[
h[0] = 15\,\mu\mathrm{m}, \quad t_{\mathrm{hold}} = 4\,\mathrm{s}, \quad t_{\mathrm{scan}} = 11\,\mathrm{s}, \quad
\sqrt{\Pfix} = 2^{\circ}, \quad \Delta\bar{u}_{\mathrm{scan}} = 0.011, \quad \text{no control of } \bar{u}_d, \bar{v}_d
\]
