\section{Truth}

$\bigl(c_\perp$ published wall curve; env $= [\bw_{lo},\ \bw_{hi}]$ planned envelope; $\bw_0$ start height; $w_c$ contact height$\bigr)$

\begin{align*}
\ba_{\mathrm{true}}(\bw) &= \frac{1}{c_\perp(\bw)} \\
b_{\mathrm{true}}(\bw)   &:= \frac{\ba'_{\mathrm{true}}(\bw)}{\bigl(1-\ba_{\mathrm{true}}(\bw)\bigr)^{2}} \\
\frac{1}{1-\ba_{\mathrm{true}}(\bw)} &= \frac{1}{1-\ba_{\mathrm{true}}(\bw_0)} + \int_{\bw_0}^{\bw} b_{\mathrm{true}}(s)\,ds \\
b'_{\mathrm{true}}(\bw)  &:= \frac{d b_{\mathrm{true}}}{d\bw}
\end{align*}

\begin{align*}

\begin{align*}
