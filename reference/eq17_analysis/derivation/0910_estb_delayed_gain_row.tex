\section{Gain law}

($e_\theta = \theta - \hat\theta$)

\begin{align*}
\ba(\bw)    &= 1 - \frac{1}{b\,(\bw - \bw_s)} \\
\ap(\ba, b) &:= \frac{d\ba}{d\bw} = b\bigl(1-\ba\bigr)^{2}
\end{align*}

\section{Control law}

$(d = 2)$

\begin{align*}
\delta\bw_3[k] &= \delta\bw_1[k] + \sum_{i=1}^{d}\Delta\bw_d[k-i] - \sum_{i=1}^{d}\ba_w[k-i]\bigl\{\fd[k-i] + \bar f_T[k-i]\bigr\}
\end{align*}

\begin{align*}
\boxed{\ \fd[k] = \ah^{-1}[k]\Bigl\{\Delta\bw_d[k] + (1-\lambda_c)\sum_{i=1}^{d}\Delta\bw_d[k-i]
       + (1-\lambda_c)\Bigl[y_1[k] - \sum_{i=1}^{d}\ah[k-i]\,\fd[k-i]\Bigr]\Bigr\}\ }
\end{align*}

\begin{align*}
y_1[k] &= \delta\bw_1[k] + n_w[k]
\end{align*}

\section{True dynamics}

\begin{align*}
\bw[k]    &= \bw_d[k] - \delta\bw_3[k] \\
\dW[k]    &:= \bw[k+1] - \bw[k] \\
\ea[k-i]  &:= \ba_w[k-i] - \ah[k-i]
\end{align*}

\begin{align*}
\delta\bw_3[k+1] &= \delta\bw_3[k] + \Delta\bw_d[k] - \ba_w[k]\bigl\{\fd[k] + \bar f_T[k]\bigr\} \tag{1} \\[4pt]
\ba_w[k]\,\fd[k] &= \ah[k]\,\fd[k] + \ea[k]\,\fd[k] \tag{2} \\[4pt]
\ah[k]\,\fd[k]   &= \Delta\bw_d[k] + (1-\lambda_c)\,\delta\bw_3[k]
                    + (1-\lambda_c)\sum_{i=1}^{d}\ea[k-i]\,\fd[k-i] \\
                 &\quad + (1-\lambda_c)\sum_{i=1}^{d}\ba_w[k-i]\,\bar f_T[k-i] + (1-\lambda_c)\,n_w[k] \tag{3} \\[4pt]
\delta\bw_3[k+1] &= \lambda_c\,\delta\bw_3[k] - \fd[k]\,\ea[k]
                    - (1-\lambda_c)\sum_{i=1}^{d}\fd[k-i]\,\ea[k-i] - \varepsilon_w[k] \tag{4}
\end{align*}

\begin{align*}
\varepsilon_w[k] &= \ba_w[k]\,\bar f_T[k] + (1-\lambda_c)\sum_{i=1}^{d}\ba_w[k-i]\,\bar f_T[k-i] + (1-\lambda_c)\,n_w[k] \\
\ba_w[k+1]       &= \ba_w[k] + b\bigl(1-\ba_w[k]\bigr)^{2}\,\dW[k] \\
b[k+1]           &= b[k]
\end{align*}

\section{Estimator}

\begin{align*}
\mathbf{x} &= \bigl[\,\delta\bw_1,\ \delta\bw_2,\ \delta\bw_3,\ \ba_w,\ b\,\bigr]^{\top}
\end{align*}

\begin{align*}
\delta\hat{\bw}_1[k+1] &= \delta\hat{\bw}_2[k] \\
\delta\hat{\bw}_2[k+1] &= \delta\hat{\bw}_3[k] \\
\delta\hat{\bw}_3[k+1] &= \lambda_c\,\delta\hat{\bw}_3[k] \\
\dWh[k]                &= \Delta\bw_d[k] + (1-\lambda_c)\,\delta\hat{\bw}_3[k] \\
\ah[k+1]               &= \ah[k] + \bh[k]\bigl(1-\ah[k]\bigr)^{2}\,\dWh[k] \\
\bh[k+1]               &= \bh[k]
\end{align*}

\section{Error dynamics}

\begin{align*}
\eD[k] &:= \dW[k] - \dWh[k]
\end{align*}

\begin{align*}
e_{\delta\bw_1}[k+1] &= e_{\delta\bw_2}[k] \\
e_{\delta\bw_2}[k+1] &= e_{\delta\bw_3}[k] \\
e_{\delta\bw_3}[k+1] &= e_{\delta\bw_3}[k] - \eD[k] \tag{5} \\
\ea[k+1]             &= \Bigl(1 - 2\bh\bigl(1-\ah[k]\bigr)\dWh[k]\Bigr)\ea[k] + \bh\bigl(1-\ah[k]\bigr)^{2}\eD[k]
                        + \bigl(1-\ah[k]\bigr)^{2}\dWh[k]\,e_b[k] \tag{6} \\
e_b[k+1]             &= e_b[k]
\end{align*}

\begin{align*}
\eD[k] &= (1-\lambda_c)\,e_{\delta\bw_3}[k] + \fd[k]\,\ea[k] + (1-\lambda_c)\sum_{i=1}^{d}\fd[k-i]\,\ea[k-i] + \varepsilon_w[k] \tag{7}
\end{align*}
