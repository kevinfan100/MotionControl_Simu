\section{Motion gain matrix as a function of the wall orientation}
\begin{align*}
\V &= \mathbf R_y(\theta_y)\,\mathbf R_x(\theta_x) = [\,\uu\ \ \vv\ \ \ww\,] \\
\Lb &= \mathrm{diag}(\aT,\ \aT,\ \aN) \\
\Ab &= \V\,\Lb\,\V^{T} = \aT\,\I + (\aN - \aT)\,\ww\,\ww^{T} \\
\Mb[k] &= \Delta\bp_d[k] + (1-\lc)\,\dpb_3[k]
\end{align*}

\section{State of the process}
\hd{Motion model, control law and dynamics of tracking error (three axes)}
\begin{align*}
\bp[k+1] &= \bp[k] + \Ab[k]\,\big(\bar{\mathbf f}_d[k] + \bar{\mathbf f}_T[k]\big) \\
\dpb[k] &= \bp_d[k] - \bp[k] \\
\bp_m[k] &= \bp[k-d] + \mathbf n_p[k] \\
\dpb_m[k] &= \bp_d[k-d] - \bp_m[k] \\
\hat{\bp}[k] &= \bp_m[k] + \sum_{i=1}^{d} \Ah[k-i]\,\bar{\mathbf f}_d[k-i] \\
\bar{\mathbf f}_d[k] &= \Ah^{-1}[k]\,\Big\{\bp_d[k+1] - \hat{\bp}[k] - \lc\,\big(\bp_d[k] - \hat{\bp}[k]\big)\Big\} \\
\dpb[k+1] &= \lc\,\dpb[k] - \big(\Ab[k] - \Ah[k]\big)\,\Fd[k] - \epp[k] \\
\Fd[k] &= \bar{\mathbf f}_d[k] + (1-\lc)\sum_{i=1}^{d} \bar{\mathbf f}_d[k-i] \\
\epp[k] &= \Ab[k]\,\bar{\mathbf f}_T[k] + (1-\lc)\sum_{i=1}^{d} \Ab[k-i]\,\bar{\mathbf f}_T[k-i] - (1-\lc)\,\mathbf n_p[k]
\end{align*}

\hd{True state of the process}
\begin{align*}
\dpb_1[k+1] &= \dpb_2[k] \\
\dpb_2[k+1] &= \dpb_3[k] \\
\dpb_3[k+1] &= \lc\,\dpb_3[k] - \big(\Ab[k] - \Ah[k]\big)\,\Fd[k] - \epp[k] \\
\aT[k+1] &= \aT[k] + \apT[k]\,\ww^{T}\Big(\Mb[k] + \big(\Ab[k] - \Ah[k]\big)\,\Fd[k] + \epp[k]\Big) \\
\apT[k+1] &= \apT[k] + w_{\apT}[k] \\
\aN[k+1] &= \aN[k] + \apN[k]\,\ww^{T}\Big(\Mb[k] + \big(\Ab[k] - \Ah[k]\big)\,\Fd[k] + \epp[k]\Big) \\
\apN[k+1] &= \apN[k] + w_{\apN}[k] \\
\theta_x[k+1] &= \theta_x[k] \\
\theta_y[k+1] &= \theta_y[k]
\end{align*}

\begin{align*}

\Ah[k] &= \Vh[k]\,\mathrm{diag}\big(\haT[k],\ \haT[k],\ \haN[k]\big)\,\Vh^{T}[k] = \haT[k]\,\I + \big(\haN[k] - \haT[k]\big)\,\wh\,\wh^{T} \\

\section{Dynamics of estimation errors}
\setline{$e_{(\cdot)} = (\cdot) - (\hat\cdot)$; $\mathbf e_3 = \dpb_3 - \dph_3$; first order in the errors}
\begin{align*}
\mathbf e[k+1] &= \mathbf F_e[k]\,\mathbf e[k] - \mathbf L[k]\,\mathbf H\,\mathbf e[k] + \mathbf q[k] \\
\mathbf q[k] &= \big[\,\mathbf 0^{T}\ \ \mathbf 0^{T}\ \ -\epp^{T}[k]\ \ \hapT\,\wh^{T}\epp[k]\ \ w_{\apT}[k]\ \ \hapN\,\wh^{T}\epp[k]\ \ w_{\apN}[k]\ \ 0\ \ 0\,\big]^{T}
\end{align*}

\hd{(a) Each factor written as its estimate plus its error}
\begin{align*}
\ww - \wh &= -\vh\,e_{\theta_x} + \cxh\,\uh\,e_{\theta_y} \\
\Mb - \Mh &= (1-\lc)\,\mathbf e_3 \\
\Ab - \Ah &= \big(\uh\,\uh^{T} + \vh\,\vh^{T}\big)\,e_{\aT} + \wh\,\wh^{T}\,e_{\aN} \\
&\quad - (\haN - \haT)\,\big(\vh\,\wh^{T} + \wh\,\vh^{T}\big)\,e_{\theta_x}
+ (\haN - \haT)\,\cxh\,\big(\uh\,\wh^{T} + \wh\,\uh^{T}\big)\,e_{\theta_y}
\end{align*}

\hd{(b) Motion along the surface normal: true minus estimated}
\begin{align*}
&\ww^{T}\Big(\Mb + \big(\Ab - \Ah\big)\,\Fd + \epp\Big) - \wh^{T}\Mh \\
&= (1-\lc)\,\wh^{T}\mathbf e_3 + \wh^{T}\Fd\,e_{\aN}
+ \Big((\haT - \haN)\,\vh^{T}\Fd - \vh^{T}\Mh\Big)\,e_{\theta_x} \\
&\quad + \cxh\,\Big(\uh^{T}\Mh - (\haT - \haN)\,\uh^{T}\Fd\Big)\,e_{\theta_y}
+ \wh^{T}\epp
\end{align*}

\section{Process outputs and output matrix}
\hd{Real-time calculation of process outputs (three axes)}
\begin{align*}
\overline{\dpb_m}[k+1] &= a_{\mathrm{pd}}\,\dpb_m[k] + (1 - a_{\mathrm{pd}})\,\overline{\dpb_m}[k] \\
\dpb_r[k] &= \dpb_m[k] - \overline{\dpb_m}[k] \\
\E\big[\dpb_r\,\dpb_r^{T}\big] &= \CK\,\Ab + C_n\,\Sn, \qquad \Sn = \mathrm{diag}\big(\sigma_{n,x}^2,\ \sigma_{n,y}^2,\ \sigma_{n,z}^2\big) \\
{}^{v}\dpb_r[k] &= \Vh^{T}[k]\,\dpb_r[k] \\
{}^{v}\Sn[k] &= \Vh^{T}[k]\,\Sn\,\Vh[k] \\
\vAm[k] &= \frac{{}^{v}\dpb_r[k]\;{}^{v}\dpb_r^{T}[k] - C_n\,{}^{v}\Sn[k]}{\CK} \\
\bar a_{Tm}[k] &= \frac{\big[\vAm[k]\big]_{uu} + \big[\vAm[k]\big]_{vv}}{2} \\
\bar a_{Nm}[k] &= \big[\vAm[k]\big]_{ww} \\
\theta_{xm}[k] &= \hat\theta_x[k] + \frac{\big[\vAm[k]\big]_{vw}}{\haT[k] - \haN[k]} \\
\theta_{ym}[k] &= \hat\theta_y[k] - \frac{\big[\vAm[k]\big]_{uw}}{\big(\haT[k] - \haN[k]\big)\,\cxh[k]}
\end{align*}
\begin{align*}
\E\big[\vAm\big] &= \Vh^{T}\,\Ab\,\Vh \\
\E\big[\vAm\big]_{uu} &\approx \aT \\
\E\big[\vAm\big]_{vv} &\approx \aT \\
\E\big[\vAm\big]_{ww} &\approx \aN \\
\E\big[\vAm\big]_{vw} &\approx (\aT - \aN)\,e_{\theta_x} \\
\E\big[\vAm\big]_{uw} &\approx -(\aT - \aN)\,\cxh\,e_{\theta_y}
\end{align*}
