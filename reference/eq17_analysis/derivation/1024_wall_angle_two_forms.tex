\part*{Part 1 \quad Estimated angle}

\section{Measurement coordinate and local coordinate}
\begin{align*}
V &= R_y(\thy)\,R_x(\theta_x) = [\,\mathbf u\ \ \mathbf v\ \ \mathbf w\,] \\
\theta_x = 0:\quad \delta\bx &= \cos\thy\,\delta\bu + \sin\thy\,\delta\bw \\
\delta\bz &= -\sin\thy\,\delta\bu + \cos\thy\,\delta\bw
\end{align*}

\section{Random component of the measured tracking error}
\begin{align*}
\overline{\dpb_m}[k+1] &= \apd\,\dpb_m[k] + (1-\apd)\,\overline{\dpb_m}[k] \\
\dpb_r[k] &= \dpb_m[k] - \overline{\dpb_m}[k] = [\,\delta\bx_r\ \ \delta\bar y_r\ \ \delta\bz_r\,]^{\mathsf T}
\end{align*}

\section{Variance and covariance in the local coordinate}
\setline{assumption: the random components along the wall and along the surface normal are uncorrelated}
\begin{align*}
\E\big[\delta\bu_r^2\big] &= \CK\,\ba_T + \Cn \\
\E\big[\delta\bw_r^2\big] &= \CK\,\ba_N + \Cn \\
\E\big[\delta\bu_r\,\delta\bw_r\big] &= 0
\end{align*}

\section{Estimated local coordinate}
\setline{the estimator uses $\hth$; the wall is at $\thy$}
\begin{align*}
e_\theta &= \thy - \hth \\
[\,\delta\tu_r\ \ \delta\tw_r\,]^{\mathsf T} &= R_y(\hth)^{\mathsf T}\,[\,\delta\bx_r\ \ \delta\bz_r\,]^{\mathsf T} \\
&= R_y(\hth)^{\mathsf T}\,R_y(\thy)\,[\,\delta\bu_r\ \ \delta\bw_r\,]^{\mathsf T} = R_y(e_\theta)\,[\,\delta\bu_r\ \ \delta\bw_r\,]^{\mathsf T} \\
\delta\tu_r &= \cos e_\theta\,\delta\bu_r + \sin e_\theta\,\delta\bw_r \\
\delta\tw_r &= -\sin e_\theta\,\delta\bu_r + \cos e_\theta\,\delta\bw_r
\end{align*}

\section{Covariance in the estimated local coordinate and the measured angle}
\begin{align*}
\delta\tu_r\,\delta\tw_r &= -\sin e_\theta\cos e_\theta\,\delta\bu_r^2 + (\cos^2 e_\theta - \sin^2 e_\theta)\,\delta\bu_r\,\delta\bw_r + \sin e_\theta\cos e_\theta\,\delta\bw_r^2 \\
\E\big[\delta\tu_r\,\delta\tw_r\big] &= -\CK\,(\ba_T - \ba_N)\,\sin e_\theta\cos e_\theta
\end{align*}
\setline{$\sin e_\theta \approx e_\theta$, $\cos e_\theta \approx 1$}
\begin{align*}
\E\big[\delta\tu_r\,\delta\tw_r\big] &\approx -\CK\,(\ba_T - \ba_N)\,e_\theta \\
e_\theta &\approx -\frac{\E\big[\delta\tu_r\,\delta\tw_r\big]}{\CK\,(\ba_T - \ba_N)}
\end{align*}
\setline{one sample in place of $\E[\,\cdot\,]$; estimates in place of $\ba_T$, $\ba_N$}
\begin{align*}
e_{\theta m}[k] &= -\frac{\delta\tu_r[k]\,\delta\tw_r[k]}{\CK\,\big(\hat{\ba}_T[k] - \hat{\ba}_N[k]\big)} \\
\theta_{ym}[k] &= \hth[k] + e_{\theta m}[k]
\end{align*}

\section*{Three axes}
\begin{align*}
[\,\delta\tu_r\ \ \delta\tv_r\ \ \delta\tw_r\,]^{\mathsf T} &= \hat V^{\mathsf T}\,\dpb_r \\
\theta_{xm}[k] &= \hat\theta_x[k] + \frac{\delta\tv_r[k]\,\delta\tw_r[k]}{\CK\,\big(\hat{\ba}_T[k] - \hat{\ba}_N[k]\big)} \\
\theta_{ym}[k] &= \hth[k] - \frac{\delta\tu_r[k]\,\delta\tw_r[k]}{\CK\,\big(\hat{\ba}_T[k] - \hat{\ba}_N[k]\big)\cos\hat\theta_x[k]}
\end{align*}

\part*{Part 2 \quad Computed angle}
\setcounter{section}{0}

\section{Motion gain matrix in the measurement coordinate}
\begin{align*}
\Ab &= \ba_T\,I + (\ba_N - \ba_T)\,\mathbf w\,\mathbf w^{\mathsf T} \\
\mathbf w &= [\,\sin\thy\cos\theta_x\ \ \ -\sin\theta_x\ \ \ \cos\thy\cos\theta_x\,]^{\mathsf T}
\end{align*}

\section{Variance and covariance in the measurement coordinate}
\setline{sensor noise taken equal on $x$, $y$ and $z$}
\begin{align*}
\E\big[\delta\bx_r\,\delta\bz_r\big] &= \CK\,\Ab_{xz} \\
\E\big[\delta\bx_r^2\big] - \E\big[\delta\bz_r^2\big] &= \CK\,\big(\Ab_{xx} - \Ab_{zz}\big) \\
\E\big[\delta\bar y_r\,\delta\bz_r\big] &= \CK\,\Ab_{yz} \\
\E\big[\delta\bar y_r^2\big] - \E\big[\delta\bz_r^2\big] &= \CK\,\big(\Ab_{yy} - \Ab_{zz}\big)
\end{align*}

\section{Wall angle}
\setline{$\theta_x = 0$}
\begin{align*}
\Ab_{xx} &= \ba_T\cos^2\thy + \ba_N\sin^2\thy \\
\Ab_{zz} &= \ba_T\sin^2\thy + \ba_N\cos^2\thy \\
\Ab_{xz} &= -(\ba_T - \ba_N)\,\sin\thy\cos\thy \\
\Ab_{xx} - \Ab_{zz} &= (\ba_T - \ba_N)\,\big(\cos^2\thy - \sin^2\thy\big) = (\ba_T - \ba_N)\,\cos 2\thy \\
-2\,\Ab_{xz} &= (\ba_T - \ba_N)\,2\sin\thy\cos\thy = (\ba_T - \ba_N)\,\sin 2\thy \\
\frac{-2\,\Ab_{xz}}{\Ab_{xx} - \Ab_{zz}} &= \frac{\sin 2\thy}{\cos 2\thy} = \tan 2\thy
\end{align*}
\setline{$\thy$ is not small; no approximation}
\begin{align*}
\tan 2\thy &= \frac{-2\,\Ab_{xz}}{\Ab_{xx} - \Ab_{zz}} \\
\tan 2\varphi_{yz} &= \frac{-2\,\Ab_{yz}}{\Ab_{yy} - \Ab_{zz}} \\
\theta_x &= -\arctan\big(\tan\varphi_{yz}\,\cos\thy\big)
\end{align*}
\enlargethispage{6\baselineskip}
\setline{estimates $\Ah$ in place of $\Ab$}
\begin{align*}
\tan 2\hth &= \frac{-2\,\Ah_{xz}}{\Ah_{xx} - \Ah_{zz}} \\
\tan 2\hat\varphi_{yz} &= \frac{-2\,\Ah_{yz}}{\Ah_{yy} - \Ah_{zz}} \\
\hat\theta_x &= -\arctan\big(\tan\hat\varphi_{yz}\,\cos\hth\big)
\end{align*}
