\begin{center}{\large\textbf{Part III\quad Regulation of the motion gain without the limit on $\Delta\bx_d$}}\end{center}

\section{Contact with the wall, $\tau_a = 0.1$ s}

\hd{Regulation law}
\[
\begin{aligned}
\Delta\bx_d[k] &= (1-\lambda_a)\,\frac{\ba_d - \hat{\ba}[k]}{\hat{\ba}'[k]} \\
\bx_d[k+1] &= \bx_d[k] + \Delta\bx_d[k] \\
\lambda_a &= 1 - T_s/\tau_a
\end{aligned}
\]
