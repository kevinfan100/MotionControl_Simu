\section{Gain law}

(bars denote normalized quantities: lengths by $R$, $\ba = a/a_o$,
$\bar{f} = a_o f$; $0 < \ba < 1$; $e_{\theta} = \theta - \hat{\theta}$)

\begin{equation*}
\ba_{\mathrm{true}}(\bw) = 1 - \frac{9}{8\bw} + \frac{1}{2\bw^{3}}
  - \frac{57}{100\,\bw^{4}} + \frac{1}{5\bw^{5}}
  + \frac{7}{200\,\bw^{11}} - \frac{1}{25\,\bw^{12}}
\end{equation*}

\begin{equation*}
\ba'_{\mathrm{true}}(\bw) > 0 \ \ \text{on} \ \ \bw > 1
\qquad\Longrightarrow\qquad
\ba' = b_{\mathrm{true}}(\ba)\bigl(1-\ba\bigr)^{2}
\ \ \text{exactly},
\qquad
b_{\mathrm{true}} := \frac{\ba'_{\mathrm{true}}}
  {\bigl(1-\ba_{\mathrm{true}}\bigr)^{2}}
\end{equation*}

\begin{align*}
\ba_{\mathrm{true}}(1) = 0, \quad \ba'_{\mathrm{true}}(1) = 1
  &\quad\Longrightarrow\quad b_{\mathrm{true}}\bigl(\ba = 0\bigr) = 1 \\
1 - \ba_{\mathrm{true}} = \frac{9}{8\bw} + \mathcal{O}(\bw^{-3})
  &\quad\Longrightarrow\quad b_{\mathrm{true}}\bigl(\ba \to 1\bigr) = \frac{8}{9}
\end{align*}

The slope is evaluated at a HEIGHT, not at the gain state. Integrating the law
once gives the level it is evaluated through:

\begin{align*}
\ba(u) &= 1 - \frac{1}{b\,u} \\
\ba'(u) &= b\bigl(1-\ba(u)\bigr)^{2} = \frac{1}{b\,u^{2}}
\end{align*}

\begin{align*}
u[k] &= \bw_d[k] - \bw_0 - \delta\hat{\bw}_3[k]
  \qquad \text{(\texttt{act}: the estimated ACTUAL height)} \\
u[k] &= \bw_d[k] - \bw_0
  \qquad\qquad\quad\ \ \text{(\texttt{cmd}: the COMMANDED height)}
\end{align*}

\begin{equation*}
a_w = a_o\,\ba_w , \qquad a_o = \frac{\Delta t}{\gamma_N R}
\end{equation*}

\section*{Observability}

Gate 1. Both routes for $b$:

\begin{align*}
\text{state route} \quad
  \frac{\partial \ba_w[k+1]}{\partial b}
    &= -\frac{1}{\hat{b}^{2}\hat{u}^{2}}
       \bigl(\Delta\bw_d + (1-\lambda_c)\delta\hat{\bw}_3\bigr) \\
\text{measurement route} \quad
  \frac{\partial y_2}{\partial b}
    &= +\frac{1}{\hat{b}^{2}\hat{u}^{2}}\,\nabla_d\bw_d
\end{align*}

Gate 2. Time signatures on the gain channel:

\begin{align*}
e_b &\ \longrightarrow\ \text{scaled by the commanded DISPLACEMENT} \\
e_{\delta\bw_3} &\ \longrightarrow\ \text{scaled by the commanded displacement AND by } 1/\hat{u} \\
e_{\ba_w} &\ \longrightarrow\ \text{scaled by the applied force } F_{dw}
\end{align*}

Gate 3. Reduction on $(\ba_w, b)$, with
$h := 1$ and $\Sigma := \Delta\bw_d + (1-\lambda_c)\delta\hat{\bw}_3$:

\begin{align*}
A_2 &= \begin{bmatrix}
  1 + \dfrac{F_{dw}}{\hat{b}\hat{u}^{2}}
  & -\dfrac{\Sigma}{\hat{b}^{2}\hat{u}^{2}} \\[0.9em]
  0 & 1
\end{bmatrix} ,
\qquad
C_2 = \begin{bmatrix} 1 & \dfrac{\nabla_d\bw_d}{\hat{b}^{2}\hat{u}^{2}} \end{bmatrix} \\[0.8em]
\det O_2 &= -\frac{\Sigma}{\hat{b}^{2}\hat{u}^{2}}
  - \frac{\nabla_d\bw_d}{\hat{b}^{2}\hat{u}^{2}}\,
    \frac{F_{dw}}{\hat{b}\hat{u}^{2}} \\[0.4em]
\det O_2 = 0 &\iff
  \Sigma = -\,\frac{\nabla_d\bw_d\,F_{dw}}{\hat{b}\hat{u}^{2}}
\end{align*}

In a hold both $\Sigma$ and $\nabla_d\bw_d$ vanish and the condition is met
identically: $b$ carries no information there, in this writing as in the
state writing.
