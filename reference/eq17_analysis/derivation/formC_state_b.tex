\section{Gain law}

(bars denote normalized quantities: lengths by $R$, $\ba = a/a_o$,
$\bar{f} = a_o f$; $0 < \ba < 1$; $e_{\theta} = \theta - \hat{\theta}$)

\begin{equation*}
\ba_{\mathrm{true}}(\bw) = 1 - \frac{9}{8\bw} + \frac{1}{2\bw^{3}}
  - \frac{57}{100\,\bw^{4}} + \frac{1}{5\bw^{5}}
  + \frac{7}{200\,\bw^{11}} - \frac{1}{25\,\bw^{12}}
\end{equation*}

\begin{equation*}
\ba'_{\mathrm{true}}(\bw) > 0 \ \ \text{on} \ \ \bw > 1
\qquad\Longrightarrow\qquad
\ba' = \frac{\bigl(1-\ba\bigr)^{2}}{b_{\mathrm{true}}(\ba)}
\ \ \text{exactly},
\qquad
b_{\mathrm{true}} := \frac{\bigl(1-\ba_{\mathrm{true}}\bigr)^{2}}
  {\ba'_{\mathrm{true}}}
\end{equation*}

\begin{align*}
\ba_{\mathrm{true}}(1) = 0, \quad \ba'_{\mathrm{true}}(1) = 1
  &\quad\Longrightarrow\quad b_{\mathrm{true}}\bigl(\ba = 0\bigr) = 1 \\
1 - \ba_{\mathrm{true}} = \frac{9}{8\bw} + \mathcal{O}(\bw^{-3})
  &\quad\Longrightarrow\quad b_{\mathrm{true}}\bigl(\ba \to 1\bigr) = \frac{9}{8}
\end{align*}

\begin{align*}
\ba'(\ba,b) &:= \frac{\bigl(1-\ba\bigr)^{2}}{b} \\
a_w &= a_o\,\ba_w , \qquad a_o = \frac{\Delta t}{\gamma_N R}
\end{align*}

\section{Jacobians}

\begin{align*}
\frac{\partial \ba'}{\partial \ba} &= -\frac{2\bigl(1-\ba\bigr)}{b} \\
\frac{\partial \ba'}{\partial b}   &= -\frac{\bigl(1-\ba\bigr)^{2}}{b^{2}}
  = -\frac{\ba'}{b}
\end{align*}

\section{True dynamics}

$\bigl(\delta\bw_j[k] := \delta\bw[k-d+j-1],\ j = 1,\dots,d+1,\ d = 2\bigr)$

\begin{align*}
\bw[k]           &= \bw_d[k] - \delta\bw_3[k] \\
\bw_d[k+1]       &= \bw_d[k] + \Delta\bw_d[k] \\
\delta\bw_1[k+1] &= \delta\bw_2[k] \\
\delta\bw_2[k+1] &= \delta\bw_3[k] \\
\delta\bw_3[k+1] &= \lambda_c\,\delta\bw_3[k] - F_{dw}[k]\,e_{\ba_w}[k]
                    - \varepsilon_w[k] \\
\ba_w[k+1]       &= \ba_w[k] + \ba'_{\mathrm{true}}[k]\Bigl[
                    \Delta\bw_d[k] + (1-\lambda_c)\,\delta\bw_3[k]
                    + F_{dw}[k]\,e_{\ba_w}[k] + \varepsilon_w[k]\Bigr] \\
b[k+1]           &= b[k]
\end{align*}

\begin{align*}
F_{dw}[k] &= \bar{f}_{dw}[k] + (1-\lambda_c)\sum_{i=1}^{d}\bar{f}_{dw}[k-i] \\
\varepsilon_w[k] &= \ba_w[k]\,\bar{f}_T[k]
  + (1-\lambda_c)\sum_{i=1}^{d}\ba_w[k-i]\,\bar{f}_T[k-i]
  + (1-\lambda_c)\,n_w[k]
\end{align*}

\section{Process model}

\begin{equation*}
\mathbf{x} = \bigl[\,\delta\bw_1,\ \delta\bw_2,\ \delta\bw_3,\ \ba_w,\ b\,\bigr]^{\top}
\qquad (d = 2)
\end{equation*}

\section{Estimator}

\begin{align*}
\delta\hat{\bw}_1[k+1] &= \delta\hat{\bw}_2[k] \\
\delta\hat{\bw}_2[k+1] &= \delta\hat{\bw}_3[k] \\
\delta\hat{\bw}_3[k+1] &= \lambda_c\,\delta\hat{\bw}_3[k] \\
\hat{\ba}_w[k+1]       &= \hat{\ba}_w[k]
  + \frac{\bigl(1-\hat{\ba}_w[k]\bigr)^{2}}{\hat{b}[k]}
    \Bigl[\Delta\bw_d[k] + (1-\lambda_c)\,\delta\hat{\bw}_3[k]\Bigr] \\
\hat{b}[k+1]           &= \hat{b}[k]
\end{align*}

\section{Error dynamics}

(True $-$ Estimator; coefficients at the estimate)

\begin{align*}
e_{\delta\bw_1}[k+1] &= e_{\delta\bw_2}[k] \\
e_{\delta\bw_2}[k+1] &= e_{\delta\bw_3}[k] \\
e_{\delta\bw_3}[k+1] &= \lambda_c\,e_{\delta\bw_3}[k]
  - F_{dw}[k]\,e_{\ba_w}[k] - \varepsilon_w[k] \\
e_{\ba_w}[k+1] &= \Bigl[\,1
  + \frac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}}F_{dw}
  - \frac{2\bigl(1-\hat{\ba}_w\bigr)}{\hat{b}}
    \bigl(\Delta\bw_d + (1-\lambda_c)\delta\hat{\bw}_3\bigr)\Bigr] e_{\ba_w}[k] \\
  &\qquad + \frac{(1-\lambda_c)\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}}
      e_{\delta\bw_3}[k]
    - \frac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}^{2}}
      \bigl(\Delta\bw_d + (1-\lambda_c)\delta\hat{\bw}_3\bigr) e_b[k] \\
  &\qquad + \frac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}}\varepsilon_w[k] \\
e_b[k+1] &= e_b[k]
\end{align*}

\begin{equation*}
\small
F_e[k] =
\begin{bmatrix}
0 & 1 & 0 & 0 & 0 \\[0.4em]
0 & 0 & 1 & 0 & 0 \\[0.4em]
0 & 0 & \lambda_c & -F_{dw} & 0 \\[0.8em]
0 & 0 & \dfrac{(1-\lambda_c)\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}}
  & \begin{array}{@{}l@{}}
      1 + \dfrac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}}F_{dw} \\[0.6em]
      \quad - \dfrac{2\bigl(1-\hat{\ba}_w\bigr)}{\hat{b}}
        \bigl(\Delta\bw_d + (1-\lambda_c)\,\delta\hat{\bw}_3\bigr)
    \end{array}
  & \begin{array}{@{}l@{}}
      -\dfrac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}^{2}} \\[0.6em]
      \quad \times \bigl(\Delta\bw_d + (1-\lambda_c)\,\delta\hat{\bw}_3\bigr)
    \end{array} \\[1.6em]
0 & 0 & 0 & 0 & 1
\end{bmatrix}
\end{equation*}

\begin{equation*}
\mathbf{q}[k] = \Bigl[\,0,\ 0,\ -\varepsilon_w,\
\frac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}}\varepsilon_w,\ 0\,\Bigr]^{\top}
\end{equation*}

\section{Measurement and innovation}

\begin{align*}
\nabla_d\bw_d[k] &= \bw_d[k] - \bw_d[k-d] \\
y_1[k] &= \delta\bw_1[k] + n_w[k] \\
y_2[k] &= \ba_w[k-d] + n_a[k]
\end{align*}

\begin{equation*}
\ba_w[k-d] = \ba_w[k]
  - \frac{\bigl(1-\ba_w[k]\bigr)^{2}}{b}\nabla_d\bw_d[k]
\end{equation*}

\begin{align*}
\frac{\partial y_2}{\partial \ba_w}
  &= 1 + \frac{2\bigl(1-\hat{\ba}_w\bigr)}{\hat{b}}\nabla_d\bw_d \\
\frac{\partial y_2}{\partial b}
  &= \frac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}^{2}}\nabla_d\bw_d
\end{align*}

\begin{equation*}
H[k] =
\begin{bmatrix}
1 & 0 & 0 & 0 & 0 \\
0 & 0 & 0 & 1 + \dfrac{2(1-\hat{\ba}_w)}{\hat{b}}\nabla_d\bw_d
  & \dfrac{(1-\hat{\ba}_w)^{2}}{\hat{b}^{2}}\nabla_d\bw_d
\end{bmatrix}
\end{equation*}

\begin{equation*}
K[k] = P^f H^{\top}\bigl(HP^fH^{\top}+R\bigr)^{-1}
\end{equation*}

\begin{align*}
e_{y_1}[k] &= e_{\delta\bw_1}[k] + n_w[k] \\
e_{y_2}[k] &= \Bigl[1 + \frac{2\bigl(1-\hat{\ba}_w\bigr)}{\hat{b}}
    \nabla_d\bw_d\Bigr] e_{\ba_w}[k]
  + \frac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}^{2}}\nabla_d\bw_d\;e_b[k]
  + n_a[k]
\end{align*}

\section{Process and measurement noise}

\begin{align*}
q_3 &= -\ba_w\,\bar{f}_T \\
q_4 &= -\frac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}}\,q_3 \\
\gamma(\bw) &= \gamma_N c(\bw) \\
\sigma^{2}_{\bar{f}_T} &= a_o^{2}\,\frac{4k_BT\gamma(\bw)}{\Delta t}
\end{align*}

\begin{align*}
Q_{33} &= \frac{4k_BT\,a_o}{R}\,\ba_w \\
Q_{34} = Q_{43} &= -\frac{\bigl(1-\hat{\ba}_w\bigr)^{2}}{\hat{b}}\,Q_{33} \\
Q_{44} &= \frac{\bigl(1-\hat{\ba}_w\bigr)^{4}}{\hat{b}^{2}}\,Q_{33} \\
Q_{55} &= 0
\end{align*}

\begin{align*}
R &= \mathrm{diag}(R_1, R_2) \\
R_1 &= \sigma^{2}_{n_w} \\
R_2 &= K_{\mathrm{var}} IF_{\mathrm{var}}\bigl(\ba_w+\xi\bigr)^{2} + d\,Q_{44} \\
K_{\mathrm{var}} &= \frac{2a_{\mathrm{cov}}}{2-a_{\mathrm{cov}}} \\
\sigma^{2}_{\delta\bw_r} &= C_{dpmr}\frac{4k_BT\,a_o}{R}\bigl(\ba_w+\xi\bigr) \\
\xi &= \frac{C_n\sigma^{2}_{n_w}}{C_{dpmr}\bigl(4k_BT a_o/R\bigr)}
\end{align*}

\section*{Seed and prior}

\begin{align*}
\ba(\bw) &= 1 - \frac{b}{\bw-\bw_s} \\
\frac{\partial \ba}{\partial b}\bigg|_{\bw}
  &= -\frac{1}{\bw-\bw_s} = -\frac{1-\ba}{b} \\
\frac{\partial \ba}{\partial \bw_s}\bigg|_{\bw} &= -\,\ba'
\end{align*}

\begin{align*}
\hat{b}[0] &= \tfrac{1}{2}\Bigl[\inf_{\bw\in\mathcal{E}} b_{\mathrm{true}}
  + \sup_{\bw\in\mathcal{E}} b_{\mathrm{true}}\Bigr] \\
P_{55}[0] = P_{bb} &= \Bigl[\tfrac{1}{2}\Bigl(\sup_{\bw\in\mathcal{E}}
  b_{\mathrm{true}} - \inf_{\bw\in\mathcal{E}} b_{\mathrm{true}}\Bigr)\Bigr]^{2} \\
\hat{\ba}_w[0] &= 1 - \frac{\hat{b}[0]}{\bw[0]-\hat{\bw}_s}
\end{align*}

\begin{align*}
P_{44}[0] &= \bigl(\hat{\ba}'[0]\bigr)^{2}P_{\bw_s}
  + \left(\frac{1-\hat{\ba}_w[0]}{\hat{b}[0]}\right)^{2}P_{bb}
  + \mathrm{floor}_a^{2} \\
P_{45}[0] &= -\,\frac{1-\hat{\ba}_w[0]}{\hat{b}[0]}\,P_{bb}
\end{align*}

\begin{align*}
e_{\ba}(\bw)\big|_{\bw_s} &= -\,\ba'(\bw)\,e_{\bw_s}
  \ \propto\ \frac{b}{(\bw-\bw_s)^{2}} \\
e_{\ba}(\bw)\big|_{b}     &= -\frac{e_b}{\bw-\bw_s}
\end{align*}
