\section{Law: slope only}

($h := \bw - \bw_s$, $\bw_s$ = visible surface, given; $\gs := h + \dl$ the shifted gap; $\dl > 0$ = effective wall below the visible surface;
$p = 2$ pinned; $\dl$, $c$ = the two constants; used on $h \ge 1.1$)

\begin{align*}
\ap(h;\,\dl,\,c)                        &= \frac{1 - c}{(h + \dl)^{2}} \\[2pt]
\frac{\partial\ap}{\partial h}          &= -\,\frac{2\,\ap}{h + \dl} \\[2pt]
\frac{\partial\ap}{\partial\dl}         &= -\,\frac{2\,\ap}{h + \dl} \qquad = \frac{\partial\ap}{\partial h} \\[2pt]
\frac{\partial\ap}{\partial c}          &= -\,\frac{1}{(h + \dl)^{2}} \\[2pt]
\frac{\partial\ap}{\partial\ba_w}       &= 0
\end{align*}

(integrated once, for the seed container only, $\ba(\infty) = 1$)
\begin{align*}
\int_{h}^{\infty}\ap\,dh'               &= \frac{1 - c}{h + \dl} \\
\ba(h;\,\dl,\,c)                        &= 1 - \frac{1 - c}{h + \dl} = \frac{h + \dl - 1 + c}{h + \dl}
                                           \qquad\Bigl(= \frac{h + c_0}{h + d_1},\ \ c_0 = \dl - 1 + c,\ \ d_1 = \dl\Bigr) \\
\frac{\partial\ba}{\partial\dl}         &= \frac{1 - c}{(h + \dl)^{2}} = \ap \\
\frac{\partial\ba}{\partial c}          &= \frac{1}{h + \dl}
\end{align*}

(family residual on a wall with constants $(\dl_w, c_w)$, and the gain error it leaves along a descent from $h_0$ if never corrected)
\begin{align*}
r'(h)   &:= \ap_{\mathrm{true}}(h) - \ap(h;\,\dl_w,\,c_w) \\
E(h)    &:= \int_{h}^{h_0} r'(h')\,dh'
\end{align*}
