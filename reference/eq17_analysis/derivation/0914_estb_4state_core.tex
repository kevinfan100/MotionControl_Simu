\section{Gain law}

(bars denote normalized quantities: lengths by $R$; $a_o := \Delta t/(\gamma_N R)$ [1/pN] and
$a_{\mathrm{nom}} := \Delta t/\gamma_N = a_o R$ [$\mu$m/pN]; $\ba := a/a_{\mathrm{nom}}$; forces by $a_o$:
$\bar f := a_o f$, so that $\ba\,\bar f = \dW$; $0 \le \ba \le 1$, far field 1, contact 0;
hats sit on the normalized quantities $\bh$, $\delta\hat{\bw}$; $e_\theta = \theta - \hat\theta$; $\bw_s$ known)

\begin{align*}
\ba(\bw, b)   &= 1 - \frac{1}{b\,(\bw - \bw_s)} \\
\ap(\ba, b)   &:= \frac{\partial\ba}{\partial\bw} = b\bigl(1-\ba\bigr)^{2} \\
\Bigl(\frac{\partial\ba}{\partial b}\Bigr)_{\bw,\,\bw_s} &= \frac{1-\ba}{b} \\
a_w &= a_o\,\ba_w , \qquad a_o = \frac{\Delta t}{\gamma_N R}
\end{align*}

\section{True dynamics}

$\bigl(\delta\bw_j[k] := \delta\bw[k-d+j-1],\ j = 1,\dots,d+1,\ d = 2\bigr)$

\begin{align*}
\bw[k]           &= \bw_d[k] - \delta\bw_3[k] \\
\bw_d[k+1]       &= \bw_d[k] + \Delta\bw_d[k] \\
\ba_w[k]         &= 1 - \frac{1}{b\,\bigl(\bw[k] - \bw_s\bigr)} \\
\delta\bw_1[k+1] &= \delta\bw_2[k] \\
\delta\bw_2[k+1] &= \delta\bw_3[k] \\
\delta\bw_3[k+1] &= \lambda_c\,\delta\bw_3[k] - F_{dw}[k]\,e_{\ba_w}[k] - \varepsilon_w[k] \\
\dW[k]           &= \Delta\bw_d[k] + (1-\lambda_c)\,\delta\bw_3[k] + F_{dw}[k]\,e_{\ba_w}[k] + \varepsilon_w[k] \\
b[k+1]           &= b[k]
\end{align*}

\begin{align*}
F_{dw}[k]        &= \bar f_{dw}[k] + (1-\lambda_c)\sum_{i=1}^{d}\bar f_{dw}[k-i] \\
\varepsilon_w[k] &= \ba_w[k]\,\bar f_T[k] + (1-\lambda_c)\sum_{i=1}^{d}\ba_w[k-i]\,\bar f_T[k-i] - (1-\lambda_c)\,n_w[k]
\end{align*}
